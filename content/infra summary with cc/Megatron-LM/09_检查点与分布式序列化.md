## 1. 模块概述

Megatron-LM支持两种检查点模式：
- **传统检查点**: 每rank保存完整模型 (简单但不灵活)
- **分布式检查点**: 各rank只保存分片 (支持跨配置加载)

### 【重点】分布式检查点特性

| 特性 | 传统检查点 | 分布式检查点 | 对比nanotron |
|------|-----------|-------------|--------------|
| **存储大小** | 模型大小×DP | 模型大小/DP | 类似dist ckpt |
| **加载灵活性** | 必须相同并行配置 | 支持跨TP/PP/DP重分片 | 有限支持 |
| **内存峰值** | 高(加载完整副本) | 低(仅加载分片) | 低 |
| **元数据** | 简单metadata.json | 完整sharding metadata | 简化 |
| **严格模式** | 不支持 | StrictHandling枚举 | 基础支持 |

## 3. 核心概念与实现

### 3.1 ShardedTensor 核心类

```python
# 文件: megatron/core/dist_checkpointing/mapping.py

@dataclass
class ShardedTensor(ShardedBase):
    """分布式张量表示。
    
    捕获本地张量与全局张量之间的映射关系，
    支持跨并行配置的灵活加载。
    
    关键属性:
    - key: 全局唯一标识符
    - data: 本地张量数据 (可能为None用于加载时占位)
    - local_shape: 本地张量形状
    - global_shape: 全局张量形状
    - global_offset: 本地张量在全局张量中的起始偏移
    - axis_fragmentations: 各维度的分片数量配置
    - replica_id: 副本标识 (区分DP副本)
    """
    
    key: str
    data: Optional[torch.Tensor] = field(repr=False)
    dtype: torch.dtype
    local_shape: Tuple[int, ...]
    global_shape: Tuple[int, ...]
    global_offset: Tuple[int, ...]
    axis_fragmentations: Optional[Tuple[int, ...]] = None
    replica_id: ReplicaId = 0
    
    @classmethod
    def from_rank_offsets(
        cls,
        key: str,
        data: torch.Tensor,
        *rank_offsets: Tuple[int, int, int],  # (axis, rank_offset, axis_fragm)
        **kwargs
    ):
        """从rank偏移构造ShardedTensor。
        
        示例: TP=4, rank 1的ColumnParallel权重
        - 原始W: [output, input]
        - 当前rank数据: W[output/4:output/4*2, :]  ([O/4, I])
        
        ShardedTensor.from_rank_offsets(
            key="fc1.weight",
            data=local_weight,  # shape [O/4, I]
            (0, 1, 4),  # 第0维(输出)，rank=1，总分4片
        )
        → global_shape=[O, I], global_offset=[O/4, 0]
        """
        global_offset = [0] * (data.ndim + kwargs.get('prepend_axis_num', 0))
        global_shape = list(data.shape)
        axis_fragmentations = [1] * len(global_shape)
        
        for axis, axis_rank_offset, axis_fragm in rank_offsets:
            # 计算该维度全局大小
            local_dim_size = global_shape[axis]
            global_shape[axis] = axis_fragm * local_dim_size
            
            # 计算在全局张量的偏移
            global_offset[axis] = axis_rank_offset * local_dim_size
            axis_fragmentations[axis] = axis_fragm
        
        return cls(
            key, data, data.dtype,
            tuple(data.shape),
            tuple(global_shape),
            tuple(global_offset),
            tuple(axis_fragmentations),
            **kwargs
        )
```

### 3.2 保存实现

```python
# 文件: megatron/training/checkpointing.py, core/dist_checkpointing/serialization.py

def save_checkpoint(iteration, model, optimizer, opt_param_scheduler, ...):
    """保存检查点入口。"""
    
    # 构建状态字典
    state_dict = {
        'iteration': iteration,
        'model': get_model_state_dict(model),
        'optimizer': get_optimizer_state_dict(optimizer) if optimizer else None,
        'opt_param_scheduler': opt_param_scheduler.state_dict() if opt_param_scheduler else None,
    }
    
    if args.use_dist_ckpt:
        save_dist_checkpoint(state_dict, checkpoint_dir)
    else:
        # 传统检查点
        torch.save(state_dict, checkpoint_path)

def get_model_state_dict(model):
    """获取模型的ShardedStateDict。"""
    if isinstance(model, list):  # PP场景
        sharded_state_dict = {}
        for pp_rank, model_chunk in enumerate(model):
            chunk_state_dict = model_chunk.sharded_state_dict()
            # 合并各PP stage的状态
            sharded_state_dict.update(chunk_state_dict)
        return sharded_state_dict
    else:
        return model.sharded_state_dict()

def save_dist_checkpoint(sharded_state_dict, checkpoint_dir):
    """保存分布式检查点。"""
    # 分离sharded tensors和普通对象
    sharded_tensors, non_sharded = preprocess_state_dict(sharded_state_dict)
    
    # 保存策略选择
    if args.dist_ckpt_format == 'zarr':
        strategy = ZarrSaveStrategy()
    else:
        strategy = FullyParallelSaveStrategy()  # 默认并行保存
    
    # 并行保存各tensor分片
    for key, sharded_tensor in sharded_tensors.items():
        shard_path = get_shard_path(checkpoint_dir, sharded_tensor, torch.distributed.get_rank())
        strategy.save_tensor(sharded_tensor.data, shard_path)
```

### 3.3 跨配置加载与重分片

```python
# 文件: megatron/core/dist_checkpointing/serialization.py

def load(sharded_state_dict, checkpoint_dir, strict=StrictHandling.ASSUME_OK_UNEXPECTED):
    """加载分布式检查点。
    
    支持跨并行配置加载是核心特性。
    """
    # 1. 加载检查点元数据
    ckpt_metadata = load_sharded_metadata(checkpoint_dir)
    
    # 2. 确定需要执行的重分片操作
    # 对比当前sharded_state_dict的global_shape和checkpoint的metadata
    
    loaded_tensors = {}
    for key, target_sharded in sharded_state_dict.items():
        # 找到源tensor的metadata
        source_info = ckpt_metadata.get(key)
        
        if source_info is None:
            if strict == StrictHandling.RAISE_ALL:
                raise KeyError(f"Key {key} not found in checkpoint")
            continue
        
        # 计算需要加载哪些源分片以构造目标分片
        required_shards = compute_required_shards(
            target_global_offset=target_sharded.global_offset,
            target_local_shape=target_sharded.local_shape,
            source_metadata=source_info,
        )
        
        # 并行加载并合并
        result = torch.empty(target_sharded.local_shape, dtype=target_sharded.dtype)
        for shard_info in required_shards:
            # 从检查点加载源分片数据
            source_data = load_tensor(shard_info.path)
            
            # 计算需要复制的slice
            target_slice, source_slice = compute_slice_overlap(
                target_sharded, shard_info
            )
            
            # 复制数据
            result[target_slice] = source_data[source_slice]
            
        loaded_tensors[key] = result
    
    return loaded_tensors

def compute_required_shards(target_info, source_metadata):
    """计算构造目标分片需要加载的源分片。
    
    算法: 在各维度上求目标range与源分片range的交集
    """
    required = []
    
    # 每个源分片 contiguous 存储
    for source_shard in source_metadata.shards:
        # 检查是否与目标区域有交集
        if ranges_overlap(
            (target_info.global_offset, target_info.global_offset + target_info.local_shape),
            (source_shard.global_offset, source_shard.global_offset + source_shard.local_shape)
        ):
            required.append(source_shard)
    
    return required
```

### 3.4 StrictHandling严格模式

```python
# 文件: megatron/core/dist_checkpointing/strict_handling.py

class StrictHandling(Enum):
    """严格处理枚举，定义加载时键不匹配的应对策略。
    
    生产场景经常需要加载部分权重的checkpoint，
    或处理模型结构微调的兼容性。
    """
    ASSUME_OK_UNEXPECTED = auto()  # 最高性能，假设无意外键
    LOG_UNEXPECTED = auto()        # 记录警告但不阻止
    LOG_ALL = auto()               # 记录所有不匹配
    RAISE_UNEXPECTED = auto()      # 有意外键时抛出异常
    RAISE_ALL = auto()             # 任何不匹配都抛出
    RETURN_ALL = auto()            # 返回缺失/意外键集合，不抛出
```
