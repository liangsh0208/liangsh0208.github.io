## 概述

Megatron-Bridge 的检查点系统提供了完整的检查点保存和加载功能，支持分布式训练的各种并行策略（TP/PP/DP/EP/CP），支持多种检查点格式，并提供异步保存、非持久化检查点等高级特性。

**源码位置**: `src/megatron/bridge/training/checkpointing.py`, `src/megatron/bridge/training/utils/checkpoint_utils.py`

```python
        """
        ...
    
    def finalize_async_saves(self, state: GlobalState, blocking: bool = False, terminate: bool = False) -> None:
        """完成异步保存操作。"""
        ...
```

### 保存流程

```python
def save_checkpoint(...):
    # 1. 确定检查点类型和保存目录
    if non_persistent_ckpt and ckpt_cfg.non_persistent_ckpt_type == "local":
        ckpt_type = CheckpointType.LOCAL
        save_dir = checkpointing_context["local_checkpoint_manager"].local_ckpt_dir
    elif non_persistent_ckpt:
        ckpt_type = CheckpointType.GLOBAL
        save_dir = ckpt_cfg.non_persistent_global_ckpt_dir or os.path.join(save_dir, "non_persistent")
        cleanup_old_non_persistent_checkpoint(save_dir, leave_ckpt_num=1)
    else:
        ckpt_type = CheckpointType.GLOBAL
    
    # 2. 收集 RNG 状态
    rng_state = get_rng_state(
        data_parallel_random_init=cfg.rng.data_parallel_random_init,
        ckpt_format=ckpt_cfg.ckpt_format,
        pg_collection=pg_collection,
        module_name=module_name,
    )
    
    # 3. 收集 Rerun 状态
    rerun_state_machine = get_rerun_state_machine()
    rerun_state = rerun_state_machine.state_dict(
        data_iterator=train_data_iterator,
        ckpt_format=ckpt_cfg.ckpt_format,
    )
    
    # 4. 生成状态字典
    state_dict = generate_state_dict(
        ckpt_cfg,
        model,
        optimizer,
        opt_param_scheduler,
        rng_state,
        iteration=train_state.step,
        optim_sd_kwargs=dict(metadata=sharded_sd_metadata),
        model_sd_kwargs=dict(metadata=sharded_sd_metadata),
        rerun_state=rerun_state,
    )
    
    # 5. 应用 PEFT 过滤（仅保存适配器参数）
    if cfg.peft is not None:
        state_dict = apply_peft_adapter_filter_to_state_dict(state_dict, cfg.peft)
    
    # 6. 根据格式保存
    if ckpt_format == "fsdp_dtensor":
        # PyTorch DCP 格式
        fs_storage_writer = torch.distributed.checkpoint.FileSystemWriter(checkpoint_name)
        torch.distributed.checkpoint.save(state_dict=state_dict, storage_writer=fs_storage_writer)
    else:
        # torch_dist 格式（Megatron Core 分布式检查点）
        async_save_request = dist_checkpointing.save(
            state_dict,
            checkpoint_name,
            save_strategy,
            async_sharded_save=ckpt_cfg.async_save,
            validate_access_integrity=validate_sharding_integrity,
        )
    
    # 7. 保存训练状态和配置
    if get_rank_safe() == 0:
        torch.save(train_state_dict, train_state_local_filename)
        torch.save(train_state_dict, train_state_global_filename)
        cfg.to_yaml(config_filename)
        
        # Megatron-LM 兼容性
        with open(tracker_filename, "w") as f:
            f.write(str(step))
    
    # 8. 异步保存调度
    if ckpt_cfg.async_save:
        schedule_async_save(state, async_save_request)
```

### generate_state_dict

```python
def generate_state_dict(
    ckpt_cfg: CheckpointConfig,
    model: list[MegatronModule],
    optimizer: Optional[MegatronOptimizer],
    opt_param_scheduler: Optional[Any],
    rng_state: Optional[ShardedObject],
    iteration: Optional[int] = None,
    optim_sd_kwargs: Optional[dict[str, Any]] = None,
    model_sd_kwargs: Optional[dict[str, Any]] = None,
    rerun_state: Optional[dict[str, Any]] = None,
    *,
    pg_collection: ProcessGroupCollection | None = None,
) -> dict[str, Any]:
    """生成检查点状态字典。"""
    
    state_dict = {}
    state_dict["checkpoint_version"] = 3.0
    if iteration is not None:
        state_dict["iteration"] = iteration
    
    # 模型状态
    state_dict.update(_generate_model_state_dict(model, model_sd_kwargs, ckpt_cfg.ckpt_format))
    
    # 优化器状态
    if ckpt_cfg.save_optim:
        if optimizer is not None and not optimizer.is_stub_optimizer:
            if ckpt_cfg.ckpt_format == "torch_dist":
                state_dict["optimizer"] = optimizer.sharded_state_dict(state_dict, **optim_sd_kwargs)
            else:
                state_dict["optimizer"] = optimizer.state_dict()
        if opt_param_scheduler is not None:
            state_dict["opt_param_scheduler"] = opt_param_scheduler.state_dict()
    
    # Rerun 状态
    if rerun_state is not None:
        state_dict["rerun_state_machine"] = rerun_state
    
    # RNG 状态
    if ckpt_cfg.save_rng:
        state_dict["rng_state"] = rng_state
    
    return state_dict
```

### _generate_model_state_dict

```python
def _generate_model_state_dict(
    model: list[MegatronModule],
    model_sd_kwargs: Optional[dict[str, Any]] = None,
    ckpt_format: str = "torch_dist",
    *,
    pg_collection: ProcessGroupCollection | None = None,
) -> dict[str, ShardedStateDict]:
    """生成模型状态字典。"""
    
    state_dict = {}
    
    if len(model) == 1:
        if ckpt_format == "torch_dist":
            state_dict["model"] = model[0].sharded_state_dict(**(model_sd_kwargs or {}))
        else:  # fsdp_dtensor 等格式
            state_dict["model"] = model[0].state_dict_for_save_checkpoint()
    else:
        # 流水线并行：多个模型块
        for i in range(len(model)):
            if ckpt_format == "torch_dist":
                state_dict["model%d" % i] = model[i].sharded_state_dict(**(model_sd_kwargs or {}))
            else:
                state_dict["model%d" % i] = model[i].state_dict_for_save_checkpoint()
    
    return state_dict
```

```python
def _load_checkpoint_from_path(load_dir, state, model, optimizer, ...):
    # 1. 解析检查点路径
    if is_checkpoint_iteration_directory(load_dir):
        checkpoint_name = load_dir
    else:
        # 读取 latest_train_state.pt
        train_state_filename = get_checkpoint_train_state_filename(load_dir, prefix="latest")
        if file_exists(train_state_filename):
            train_state = read_train_state(train_state_filename)
            iteration = train_state.step
        else:
            # 兼容 Megatron-LM tracker
            iteration, release = read_metadata(get_checkpoint_tracker_filename(load_dir))
        checkpoint_name = get_checkpoint_name(load_dir, iteration, release)
    
    # 2. 确定检查点格式
    ckpt_format = _get_checkpoint_format(checkpoint_name)
    
    # 3. 根据 TP/PP 匹配决定是否加载 RNG/优化器状态
    run_config = read_run_config(config_filename)
    ckpt_tp_pp = (run_config["model"]["tensor_model_parallel_size"], run_config["model"]["pipeline_model_parallel_size"])
    run_tp_pp = (cfg.model.tensor_model_parallel_size, cfg.model.pipeline_model_parallel_size)
    tp_pp_match = ckpt_tp_pp == run_tp_pp
    
    # 4. 生成目标状态字典结构
    sharded_state_dict = generate_state_dict(cfg.checkpoint, model, optimizer, ...)
    
    # 5. 加载检查点
    state_dict = dist_checkpointing.load(sharded_state_dict, checkpoint_name, load_strategy, strict=dist_ckpt_strictness)
    
    # 6. 加载模型权重
    if len(model) == 1:
        model[0].load_state_dict(state_dict["model"], strict=strict)
    else:
        for i in range(len(model)):
            model_key = "model%d" % i
            if model_key in state_dict:
                model[i].load_state_dict(state_dict[model_key], strict=strict)
    
    # 7. 加载优化器状态
    if not release and not cfg.checkpoint.finetune and cfg.checkpoint.load_optim:
        optimizer.load_state_dict(state_dict["optimizer"])
    
    # 8. 加载 RNG 状态
    if not ignore_rng_state:
        random.setstate(rng_state["random_rng_state"])
        np.random.set_state(rng_state["np_rng_state"])
        torch.set_rng_state(rng_state["torch_rng_state"])
        torch.cuda.set_rng_state(rng_state["cuda_rng_state"])
        cuda_rng_tracker.set_states(rng_state["rng_tracker_states"])
    
    # 9. 恢复训练状态
    state.train_state = read_train_state(train_state_filename)
    
    return state.train_state.step, state.train_state.floating_point_operations_so_far
```

## RNG 状态管理

### get_rng_state

```python
def get_rng_state(
    data_parallel_random_init: bool,
    ckpt_format: str = "torch_dist",
    *,
    pg_collection: ProcessGroupCollection,
    module_name: str | None = None,
) -> ShardedObject | dict:
    """
    获取所有必要库的随机数生成器状态。
    
    收集 random、numpy、torch、cuda 和 Megatron RNG tracker 的状态。
    
    对于 torch_dist 格式和专家并行（EP > 1），RNG 状态按 (PP, TP, DP) 分片，
    因为不同的 EP rank 可能有不同的 RNG 状态。
    
    没有 EP 时，状态按 (PP, TP) 分片，DP rank 作为 replica_id。
    """
    
    rng_state = {
        "random_rng_state": random.getstate(),
        "np_rng_state": np.random.get_state(),
        "torch_rng_state": torch.get_rng_state(),
        "cuda_rng_state": torch.cuda.get_rng_state(),
        "rng_tracker_states": tensor_parallel.get_cuda_rng_tracker().get_states(),
    }
    
    # 跨 DP rank 收集
    if torch.distributed.is_initialized() and pg_collection.dp_cp.size() > 1 and data_parallel_random_init:
        rng_state_list = [None for i in range(pg_collection.dp_cp.size())]
        torch.distributed.all_gather_object(rng_state_list, rng_state, group=pg_collection.dp_cp)
    else:
        rng_state_list = [rng_state]
    
    if ckpt_format == "torch_dist":
        pp_rank = pg_collection.pp.rank()
        pp_size = pg_collection.pp.size()
        tp_rank = pg_collection.tp.rank()
        tp_size = pg_collection.tp.size()
        ep_size = get_pg_size(pg_collection.ep)
        
        key = f"rng_state.{module_name}" if module_name else "rng_state"
        
        if ep_size > 1:
            # EP 模式：按 (PP, TP, DP) 分片
            dp_rank = pg_collection.dp_cp.rank()
            dp_size = pg_collection.dp_cp.size()
            rng_state_list = ShardedObject(
                key, rng_state_list, (pp_size, tp_size, dp_size), (pp_rank, tp_rank, dp_rank), replica_id=0
            )
        else:
            # 非 EP 模式：按 (PP, TP) 分片，DP 作为 replica
            rng_state_list = ShardedObject(
                key, rng_state_list, (pp_size, tp_size), (pp_rank, tp_rank), replica_id=pg_collection.dp_cp.rank()
            )
    elif ckpt_format == "fsdp_dtensor":
        pp_rank = pg_collection.pp.rank()
        tp_rank = pg_collection.tp.rank()
        rng_state_list = {f"({pp_rank}, {tp_rank})": rng_state_list}
    
    return rng_state_list
```

```python
def finalize_async_saves(state: GlobalState, ckpt_cfg: CheckpointConfig, blocking: bool = False, terminate: bool = False):
    """完成异步保存。"""
    
    if not ckpt_cfg.async_save:
        return
    
    async_queue = state.async_calls_queue
    if async_queue is None:
        return
    
    if blocking and not is_empty_async_queue(state):
        print_rank_0("Unfinalized async checkpoint saves. Finalizing them synchronously now.")
    
    async_queue.maybe_finalize_async_calls(blocking)
    
    if terminate:
        async_queue.close()
```

### 清理旧检查点

```python
def cleanup_old_non_persistent_checkpoint(save_dir: str, leave_ckpt_num: int = 1, do_async: bool = False):
    """清理旧的非持久化检查点。"""
    
    if torch.distributed.is_initialized() and torch.distributed.get_rank() != 0:
        return
    
    save_dir = Path(save_dir)
    iter_prefix = "iter_"
    
    # 查找所有 iter_* 目录
    iter_ckpts = save_dir.rglob(f"{iter_prefix}*")
    sorted_iter_ckpts = sorted(iter_ckpts, key=lambda ckpt_name: int(ckpt_name.name[len(iter_prefix) :]))
    
    if not sorted_iter_ckpts:
        return
    
    # 保留最新的 N 个
    rm_iter_ckpts = sorted_iter_ckpts[:-leave_ckpt_num]
    print_rank_0(f"Non-persistent checkpoints scheduled for removal: {rm_iter_ckpts}")
    
    def remove_iter_ckpts(_iter_ckpts):
        for ckpt in _iter_ckpts:
            shutil.rmtree(ckpt)
    
    if do_async:
        threading.Thread(target=remove_iter_ckpts, args=(rm_iter_ckpts,)).start()
    else:
        remove_iter_ckpts(rm_iter_ckpts)
```

## PEFT 检查点过滤

```python
def apply_peft_adapter_filter_to_state_dict(state_dict: dict[str, Any], peft_config: PEFT) -> dict[str, Any]:
    """
    过滤状态字典，仅保留 PEFT 适配器参数。
    
    保留所有非模型元数据（checkpoint_version、iteration 等），
    仅过滤模型权重部分。
    """
    
    return {
        checkpoint_section_key: (
            # 过滤模型参数，仅保留适配器权重
            {
                parameter_name: parameter_value
                for parameter_name, parameter_value in checkpoint_section_value.items()
                if peft_config.adapter_key_filter(parameter_name)
            }
            if _is_model_section(checkpoint_section_key)
            else checkpoint_section_value
        )
        for checkpoint_section_key, checkpoint_section_value in state_dict.items()
    }


def _is_model_section(section_key: str) -> bool:
    """检查是否为模型部分。"""
    is_single_model = section_key == "model"
    is_pipeline_model = (
        section_key.startswith("model")
        and section_key != "model"
        and section_key[5:].isdigit()
    )
    return is_single_model or is_pipeline_model
```

```python
    # 检查 tracker 文件
    train_state_filename = os.path.join(checkpoints_path, f"latest_train_state.pt")
    if file_exists(train_state_filename):
        return True
```

```python
def save_tokenizer_assets(tokenizer: MegatronTokenizer, tokenizer_config: TokenizerConfig, checkpoint_path: str):
    """保存 tokenizer 文件到检查点目录，确保检查点自包含。"""
    if tokenizer is None:
        return
    
    rank = torch.distributed.get_rank() if torch.distributed.is_initialized() else 0
    if rank != 0:
        return
    
    tokenizer_dir = os.path.join(checkpoint_path, "tokenizer")
    os.makedirs(tokenizer_dir, exist_ok=True)
    
    tokenizer_type = tokenizer_config.tokenizer_type
    
    # HuggingFace Tokenizer
    if tokenizer_type in ("HuggingFaceTokenizer", "MultimodalTokenizer"):
        if hasattr(tokenizer, "save_pretrained"):
            tokenizer.save_pretrained(tokenizer_dir)
        return
    
    # 文件类型 Tokenizer
    if tokenizer_type in ("BertWordPieceLowerCase", "BertWordPieceCase"):
        if tokenizer_config.vocab_file:
            shutil.copy2(tokenizer_config.vocab_file, os.path.join(tokenizer_dir, "vocab.txt"))
    
    elif tokenizer_type == "GPT2BPETokenizer":
        if tokenizer_config.vocab_file:
            shutil.copy2(tokenizer_config.vocab_file, os.path.join(tokenizer_dir, "vocab.json"))
        if tokenizer_config.merge_file:
            shutil.copy2(tokenizer_config.merge_file, os.path.join(tokenizer_dir, "merges.txt"))
    
    elif tokenizer_type in ("SentencePieceTokenizer", "GPTSentencePieceTokenizer", "Llama2Tokenizer"):
        if tokenizer_config.tokenizer_model:
            shutil.copy2(tokenizer_config.tokenizer_model, os.path.join(tokenizer_dir, "tokenizer.model"))
    
    elif tokenizer_type == "TikTokenizer":
        if tokenizer_config.tokenizer_model:
            shutil.copy2(tokenizer_config.tokenizer_model, os.path.join(tokenizer_dir, "tokenizer.json"))
```

```python
def maybe_save_dataloader_state(model, train_iterator, iteration, dataloader_save_path, *, pg_collection=None):
    """保存数据加载器状态（如果支持）。"""
    if train_iterator is None or dataloader_save_path is None or dataloader_save_path == "":
        return
    
    if not hasattr(train_iterator.iterable, "save_state"):
        raise RuntimeError(f"Could not find a save_state for the train_iterator of type {type(train_iterator)}")
    
    pg_collection = pg_collection or get_pg_collection(model)
    is_first_rank = (pg_collection.pp.rank() == 0) and (pg_collection.tp.rank() == 0)
    if not is_first_rank:
        return
    
    dp_rank = pg_collection.dp.rank()
    train_dataloader_state_dict = train_iterator.iterable.save_state()
    
    iter_dir = get_checkpoint_name(dataloader_save_path, iteration)
    data_state_save_path = os.path.join(iter_dir, f"train_dataloader_dprank{dp_rank:03d}.pt")
    
    torch.distributed.barrier(group=pg_collection.dp)
    
    if pg_collection.dp.rank() == 0:
        ensure_directory_exists(data_state_save_path)
    
    torch.distributed.barrier(group=pg_collection.dp)
    
    torch.save({"dataloader_state_dict": train_dataloader_state_dict}, data_state_save_path)
```

## 检查点管理器工厂

```python
def create_checkpoint_manager(checkpoint_config: CheckpointConfig) -> CheckpointManager:
    """
    创建检查点管理器。
    
    支持通过 custom_manager_class 配置自定义实现。
    """
    if checkpoint_config.custom_manager_class is not None:
        import importlib
        
        module_path, class_name = checkpoint_config.custom_manager_class.rsplit(".", 1)
        module = importlib.import_module(module_path)
        custom_manager_class = getattr(module, class_name)
        
        manager = custom_manager_class(checkpoint_config)
        
        if not isinstance(manager, CheckpointManager):
            raise TypeError(
                f"Custom checkpoint manager '{checkpoint_config.custom_manager_class}' "
                f"does not implement the CheckpointManager protocol."
            )
        
        return manager
    
    return DefaultCheckpointManager(checkpoint_config)
```

## 总结

Megatron-Bridge 的检查点系统通过：

1. **多种格式支持**: `torch_dist`（Megatron Core 分布式检查点）、`fsdp_dtensor`（PyTorch DCP）
2. **检查点类型**: GLOBAL（持久化）、LOCAL（非持久化快速恢复）
3. **并行策略兼容**: 支持 TP/PP/DP/EP/CP 各种并行组合
4. **异步保存**: 通过 `AsyncRequest` 支持非阻塞检查点
5. **PEFT 支持**: 自动过滤仅保存适配器参数
6. **状态完整性**: 模型、优化器、调度器、RNG、数据加载器状态全覆盖
7. **Megatron-LM 兼容**: 支持读取 legacy 检查点
8. **可扩展架构**: `CheckpointManager` 协议支持自定义实现

提供了企业级分布式训练所需的完整检查点解决方案。
