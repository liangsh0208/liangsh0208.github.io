## 概述

Megatron-Bridge 的训练框架提供了完整的分布式训练循环实现，包括前向传播、反向传播、优化器更新、检查点保存、评估、日志记录等核心功能。

**源码位置**: `src/megatron/bridge/training/train.py`, `src/megatron/bridge/training/pretrain.py`

```python
def pretrain(config, forward_step_func, callbacks):
    # 1. 运行时配置更新
    runtime_config_update(config)
    
    # 2. 创建全局状态
    state = GlobalState()
    state.cfg = config
    
    # 3. 规范化回调
    callback_manager = normalize_callbacks(callbacks)
    
    # 4. 检查进程重启配置
    if config.inprocess_restart and config.inprocess_restart.enabled:
        # 使用 in-process restart wrapper
        wrapped_pretrain, store = maybe_wrap_for_inprocess_restart(_pretrain, config.inprocess_restart, state)
        wrapped_pretrain(state, forward_step_func, callback_manager, store=store)
    else:
        # 正常执行
        _pretrain(state, forward_step_func, callback_manager)
```

### _pretrain

内部训练逻辑：

```python
def _pretrain(
    state: GlobalState,
    forward_step_func: ForwardStepCallable,
    callback_manager: CallbackManager | None = None,
    store: dist.Store | None = None,
    inprocess_call_wrapper: CallWrapper | None = None,
) -> None:
    """内部训练逻辑"""
    # 1. 确定是否需要销毁进程组
    should_destroy_process_group = not dist.is_initialized()
    
    # 2. 处理 in-process restart store
    if inprocess_call_wrapper is not None:
        restart_attempt = inprocess_call_wrapper.iteration
        store = dist.PrefixStore(str(restart_attempt), store)
    
    # 3. 执行 setup
    setup_output = setup(state, dataset_provider, restart_store=store, callback_manager=callback_manager)
    
    # 4. 训练循环
    if not config.validation.skip_train:
        train(forward_step_func, model, optimizer, scheduler, ...)
    
    # 5. 验证和测试
    if state.train_state.do_valid:
        evaluate_and_print_results(...)
    if state.train_state.do_test:
        evaluate_and_print_results(..., is_test=True)
    
    # 6. 清理
    _finish_train(state, checkpoint_manager)
```

```python
    callback_manager: CallbackManager | None = None,
) -> None:
    """
    主训练循环。
    处理迭代循环、train_step 调用、评估、检查点、日志和退出条件。
    """
```

### 训练循环核心结构

```python
def train(...):
    # 初始化
    wrapped_forward_step_func = prepare_forward_step_func(forward_step_func, global_state)
    
    # 开启训练模式
    for model_module in model:
        model_module.train()
    
    # 初始化 CUDA Graph
    if model_config.cuda_graph_impl == "transformer_engine":
        cuda_graph_helper = TECudaGraphHelper(model, config, ...)
    
    # 获取 forward_backward 函数
    forward_backward_func = get_forward_backward_func(pp_size, vp_size)
    
    # 主训练循环
    while global_state.train_state.step < train_config.train_iters:
        # 1. 处理 profiling
        handle_profiling_step(...)
        
        # 2. 异步检查点完成
        checkpoint_manager.finalize_async_saves(...)
        
        # 3. 更新 microbatch 数量
        update_num_microbatches(...)
        
        # 4. CUDA Graph 捕获
        if should_capture_cuda_graph:
            cuda_graph_helper.create_cudagraphs()
        
        # 5. 执行训练步骤
        (loss_dict, skipped_iter, should_checkpoint, should_exit, ...) = 
            train_step(forward_step_func, data_iterator, model, optimizer, ...)
        
        # 6. 更新计数器
        global_state.train_state.step += 1
        global_state.train_state.consumed_train_samples += batch_size
        
        # 7. 日志记录
        training_log(loss_dict, ...)
        
        # 8. 验证
        if should_validate:
            evaluate_and_print_results(...)
        
        # 9. 检查点决策
        should_exit = checkpoint_and_decide_exit(...)
        
        if should_exit:
            break
```

## 单步训练：train_step

```python
def train_step(
    forward_step_func: ForwardStepCallable,
    data_iterator: Optional[Union[RerunDataIterator, list[RerunDataIterator]]],
    model: list[MegatronModule],
    optimizer: MegatronOptimizer,
    scheduler: OptimizerParamScheduler,
    global_state: GlobalState,
    pg_collection: ProcessGroupCollection,
    forward_backward_func: Callable,
    p2p_communicator: P2PCommunicator,
) -> tuple[dict, int, bool, bool, int, Optional[float], Optional[int]]:
    """
    单步训练。
    
    Returns:
        - loss_dict: 损失字典
        - skipped_iter: 是否跳过迭代
        - should_checkpoint: 是否保存检查点
        - should_exit: 是否退出
        - exit_code: 退出码
        - grad_norm: 梯度范数
        - num_zeros_in_grad: 梯度中零的数量
    """
```

### train_step 执行流程

```python
def train_step(...):
    rerun_state_machine = get_rerun_state_machine()
    
    while rerun_state_machine.should_run_forward_backward(data_iterator):
        # 1. 清零梯度
        for model_chunk in model:
            model_chunk.zero_grad_buffer()
        optimizer.zero_grad()
        
        # 2. 处理 finetuning 数据
        if cfg.dataset.dataloader_type == "batch":
            forward_backward_data_iterator, seq_length = prepare_finetuning_batch(...)
        
        # 3. 创建数据迭代器列表（虚拟流水线并行）
        if len(model) > 1:
            forward_backward_data_iterator = make_data_iterator_list(model, data_iterator)
        
        # 4. 前向-反向传播
        losses_reduced = forward_backward_func(
            forward_step_func=forward_step_func,
            data_iterator=forward_backward_data_iterator,
            model=model,
            num_microbatches=get_num_microbatches(),
            seq_length=seq_length,
            micro_batch_size=train_config.micro_batch_size,
            forward_only=False,
            p2p_communicator=p2p_communicator,
            pg_collection=pg_collection,
        )
    
    # 5. 清理内存
    if train_config.empty_unused_memory_level >= 1:
        torch.cuda.empty_cache()
    
    # 6. 优化器更新
    timers("optimizer").start()
    update_successful, grad_norm, num_zeros_in_grad = optimizer.step()
    timers("optimizer").stop()
    
    # 7. 更新学习率
    if update_successful:
        increment = get_num_microbatches() * train_config.micro_batch_size * dp_size
        scheduler.step(increment=increment)
        skipped_iter = 0else:
        skipped_iter = 1
    
    # 8. 计算损失
    if pp_last_stage:
        loss_reduced = {}
        for key in losses_reduced[0].keys():
            val = [x[key].view(-1) for x in losses_reduced]
            if val[0].numel() == 2:
                # 新报告方式：按 token 数平均
                val = torch.vstack(val).sum(dim=0)
                torch.distributed.all_reduce(val, group=dp_cp_group)
                loss_reduced[key] = val[0] / val[1]
            else:
                # 旧行为：按microbatch 数平均
                loss_reduced[key] = torch.cat(val).mean()
        
        return (loss_reduced, skipped_iter, should_checkpoint, ...)
    
    return ({}, skipped_iter, should_checkpoint, ...)
```

## 检查点管理

### save_checkpoint_and_time

```python
def save_checkpoint_and_time(
    state: GlobalState,
    model: list[MegatronModule],
    optimizer: MegatronOptimizer,
    opt_param_scheduler: OptimizerParamScheduler,
    num_floating_point_operations_so_far: float,
    checkpoint_manager: CheckpointManager,
    non_persistent_ckpt: bool = False,
    train_data_iterator: Optional[...] = None,
    pg_collection: Optional[ProcessGroupCollection] = None,
    callback_manager: Optional[CallbackManager] = None,
    module_name: str | None = None,
) -> None:
    """保存检查点并记录时间"""
    # 停止计时器
    timers("interval-time").stop()
    
    # 暂停能耗监控
    if energy_monitor:
        energy_monitor.pause()
    # 强制参数同步（分布式优化器）
    if should_force_param_sync:
        force_param_sync(model)
    
    # 释放内存
    for model_chunk in model:
        if hasattr(model_chunk, "free_overlap_buffers"):
            model_chunk.free_overlap_buffers()
    torch.cuda.empty_cache()
    
    # 保存检查点
    checkpoint_manager.save(
        CheckpointSaveContext(
            state=state,
            model=model,
            optimizer=optimizer,
            opt_param_scheduler=opt_param_scheduler,
            num_floating_point_operations_so_far=int(num_floating_point_operations_so_far),
            train_data_iterator=train_data_iterator,
            non_persistent_ckpt=non_persistent_ckpt,
            pg_collection=pg_collection,
            module_name=module_name,
        ),
        callback_manager,
    )
```

### checkpoint_and_decide_exit

```python
def checkpoint_and_decide_exit(
    state: GlobalState,
    model: list[MegatronModule],
    optimizer: MegatronOptimizer,
    opt_param_scheduler: OptimizerParamScheduler,
    num_floating_point_operations_so_far: float,
    checkpoint_manager: CheckpointManager,
    train_data_iterator: Optional[...] = None,
    pg_collection: Optional[ProcessGroupCollection] = None,
    callback_manager: Optional[CallbackManager] = None,
    module_name: str | None = None,
) -> bool:
    """
    处理检查点决策并确定是否退出训练。
    
    检查条件：
    - 信号处理
    - 间隔保存
    - 持续时间
    - 迭代间隔
    - Straggler 检测
    """
    # 1. 信号处理退出
    if state.cfg.train.exit_signal_handler:
        if any(signal_handler.signals_received()):
            save_checkpoint_and_time(...)
            return True
    
    # 2. 常规保存
    if state.cfg.checkpoint.save and state.train_state.step % save_interval == 0:
        save_checkpoint_and_time(...)
    
    # 3. 持续时间退出
    if state.cfg.train.exit_duration_in_mins:
        train_time = (time.time() - state.start_time) / 60.0
        if train_time > state.cfg.train.exit_duration_in_mins:
            save_checkpoint_and_time(...)
            return True
    
    # 4. 迭代间隔退出
    if state.cfg.train.exit_interval and state.train_state.step % exit_interval == 0:
        save_checkpoint_and_time(...)
        return True
    
    # 5. Straggler 检测退出
    if check_nvrx_straggler_detection(state.nvrx_straggler_manager):
        save_checkpoint_and_time(...)
        return True
    
    return False
```

## Forward Pre-Hook 管理

```python
def should_disable_forward_pre_hook(
    use_megatron_fsdp: bool, 
    use_distributed_optimizer: bool, 
    overlap_param_gather: bool
) -> bool:
    """
    确定是否在检查点期间禁用 forward pre-hook。
    当使用分布式优化器和重叠参数收集时需要禁用。
    """
    return not use_megatron_fsdp and use_distributed_optimizer and overlap_param_gather

def enable_forward_pre_hook(model: list[DDP]) -> None:
    """启用所有模型块的 forward pre-hook"""
    for model_chunk in model:
        model_chunk.enable_forward_pre_hook()

def disable_forward_pre_hook(model: list[DDP], param_sync: bool = True) -> None:
    """禁用所有模型块的 forward pre-hook"""
    for model_chunk in model:
        model_chunk.disable_forward_pre_hook(param_sync=param_sync)
```

### Straggler 检测

```python
def maybe_report_stragglers(
    log_interval: int,
    log_straggler: bool,
    straggler_timer: Any,
    iteration: int,
    num_floating_point_operations_since_last_log_event: float,
) -> float:
    """报告 straggler 指标"""
    if log_straggler and log_interval:
        if iteration % log_interval == 0:
            straggler_timer.report(
                num_floating_point_operations_since_last_log_event,
                log_interval,
            )
            return 0.0
    return num_floating_point_operations_since_last_log_event
```

```python
def maybe_check_weight_hash_across_dp_replicas(
    model: list[MegatronModule],
    check_weight_hash_across_dp_replicas_interval: Optional[int],
    iteration: int,
    should_toggle_forward_pre_hook: bool,
) -> None:
    """验证 DP 副本间的权重哈希"""
    if check_weight_hash_across_dp_replicas_interval is None:
        return
    if iteration % check_weight_hash_across_dp_replicas_interval != 0:
        return
    
    if should_toggle_forward_pre_hook:
        disable_forward_pre_hook(model)    
    assert check_param_hashes_across_dp_replicas(model, cross_check=True)
    torch.distributed.barrier()
    
    if should_toggle_forward_pre_hook:
        enable_forward_pre_hook(model)
```

### 吞吐量计算

```python
def compute_throughputs_and_append_to_progress_log(
    state: GlobalState, 
    num_floating_point_operations_so_far: float
) -> None:
    """计算并记录吞吐量"""
    # 任务吞吐量
    job_throughput = (num_floating_point_operations_so_far - state.train_state.floating_point_operations_so_far) / (
        (time.time() - state.start_time) * 10**12 * get_world_size_safe()
    )
    
    # 累积吞吐量
    start_time, start_num_floating_point_operations = get_start_time_from_progress_log(state.cfg)
    elapsed_time = (datetime.now() - start_time).total_seconds()
    cumulative_throughput = (num_floating_point_operations_so_far - start_num_floating_point_operations) / (
        elapsed_time * 10**12 * get_world_size_safe()
    )
    
    append_to_progress_log(
        state.cfg.checkpoint.save,
        f"Saved checkpoint\tIteration: {state.train_state.step}\t"
        f"Job throughput: {job_throughput:.1f} MODEL_TFLOP/s/GPU\t"
        f"Cumulative throughput: {cumulative_throughput:.1f} MODEL_TFLOP/s/GPU\t"
        f"Floating-point operations: {num_floating_point_operations_so_far:.2e}\t"
        f"Tokens (in billions): {tokens_so_far / 10**9:.2f}",
    )
```

## 清理函数

```python
def _finish_train(global_state: GlobalState, checkpoint_manager: CheckpointManager):
    """训练结束时的清理函数"""
    # 关闭 NVRx straggler 检测
    safe_shutdown_nvrx_straggler_manager(global_state.nvrx_straggler_manager)
    
    # 完成异步保存
    fault_tolerance.on_checkpointing_start(global_state)
    checkpoint_manager.finalize_async_saves(state=global_state, blocking=True, terminate=True)
    fault_tolerance.on_checkpointing_end(global_state=global_state, is_async_finalization=True)
    
    # 关闭 fault tolerance
    fault_tolerance.shutdown(global_state)
    
    # 关闭日志
    if global_state.wandb_logger:
        global_state.wandb_logger.finish()
    if global_state._comet_logger:
        global_state._comet_logger.end()
    
    # 销毁全局状态
    destroy_global_state()
```

## 总结

Megatron-Bridge 的训练框架通过：

1. **模块化设计**: `pretrain` → `train` → `train_step` 清晰分层
2. **完整功能**: 涵盖训练、验证、检查点、日志、监控
3. **高级特性**: CUDA Graph、Straggler 检测、Callback 系统
4. **容错机制**: In-process restart、信号处理、异步检查点

提供了企业级分布式训练所需的全部功能。
