# Megatron-LM 训练循环模块详解

> 训练流程第四环：前向/反向传播与参数更新

---

## 1. 模块概述

训练循环模块负责协调整个训练过程，包括数据迭代、前向传播、反向传播、梯度同步和参数更新。

```
pretrain()
    │
    ▼
setup_model_and_optimizer()   # 模型 + 优化器初始化
    │
    ▼
build_train_valid_test_data_iterators()  # 数据迭代器
    │
    ▼
train()                       # 主训练循环
    │
    ├─ while iteration < train_iters:
    │       │
    │       ▼
    │   train_step()          # 单步训练
    │       │
    │       ├── zero_grad()             # 梯度清零
    │       ├── forward_backward_func() # 前向 + 反向
    │       ├── optimizer.step()        # 参数更新
    │       └── opt_param_scheduler.step() # LR 更新
    │
    ├─ 定期 evaluate()         # 验证
    └─ 定期 save_checkpoint()  # 检查点保存
```

**核心文件**:
- `megatron/training/training.py` - 主训练逻辑
- `megatron/core/pipeline_parallel/schedules.py` - PP 调度策略

## 2. 主训练循环 (train)

**文件位置**: `megatron/training/training.py`

### 2.1 train() 函数结构

```python
def train(
    forward_step_func,      # 用户定义的前向步骤函数
    model,                  # 模型列表 (VP 时多个 chunk)
    optimizer,              # 优化器
    opt_param_scheduler,    # LR scheduler
    train_data_iterator,    # 训练数据迭代器
    valid_data_iterator,    # 验证数据迭代器
    process_non_loss_data_func,
    config,                 # TransformerConfig
    checkpointing_context,
    non_loss_data_func,
):
    """主训练循环"""
    # 1. 设置模型为训练模式
    for model_module in model:
        model_module.train()

    # 2. 配置通信重叠参数
    config.grad_scale_func = optimizer.scale_loss
    if args.overlap_grad_reduce:
        config.no_sync_func = [m.no_sync for m in model]
        config.grad_sync_func = [m.start_grad_sync for m in model]
    if args.overlap_param_gather:
        config.param_sync_func = [m.start_param_sync for m in model]

    # 3. 获取 forward_backward_func (根据 PP 配置自动选择)
    forward_backward_func = get_forward_backward_func()

    # 4. 主循环
    while iteration < args.train_iters:
        update_num_microbatches(consumed_samples)  # 更新 micro-batch 数

        # 执行单步训练
        loss_dict, skipped_iter, ... = train_step(
            forward_step_func, data_iterator, model,
            optimizer, opt_param_scheduler, config,
            forward_backward_func, iteration=iteration
        )

        # 日志记录
        training_log(loss_dict, ...)

        # 定期验证和保存检查点
        if iteration % args.eval_interval == 0:
            evaluate_and_print_results(...)
        if iteration % args.save_interval == 0:
            save_checkpoint_and_time(...)

        iteration += 1
        args.consumed_train_samples += batch_size
```

## 3. 单步训练 (train_step)

### 3.1 完整流程

```python
def train_step(
    forward_step_func,
    data_iterator,
    model,
    optimizer,
    opt_param_scheduler,
    config,
    forward_backward_func,
    iteration=None,
):
    """单步训练"""
    # 1. 梯度清零
    for model_chunk in model:
        model_chunk.zero_grad_buffer()  # 清零梯度 buffer
    optimizer.zero_grad()

    # 2. 前向 + 反向传播
    losses_reduced = forward_backward_func(
        forward_step_func=forward_step_func,
        data_iterator=data_iterator,
        model=model,
        num_microbatches=get_num_microbatches(),   # 当前 micro-batch 数
        seq_length=args.seq_length,
        micro_batch_size=args.micro_batch_size,
        forward_only=False,
    )

    # 3. 参数更新
    update_successful, grad_norm, num_zeros_in_grad = optimizer.step()

    # 4. 跨 MP rank 聚合更新状态
    update_successful = logical_and_across_model_parallel_group(update_successful)
    grad_norm = reduce_max_stat_across_model_parallel_group(grad_norm)

    # 5. 更新 LR (只有更新成功时)
    if update_successful:
        increment = get_num_microbatches() * args.micro_batch_size * args.data_parallel_size
        opt_param_scheduler.step(increment=increment)
        skipped_iter = 0
    else:
        skipped_iter = 1  # FP16 overflow 导致跳过

    # 6. 聚合 Loss (只有 PP 最后一个 stage 有完整 loss)
    if mpu.is_pipeline_last_stage(ignore_virtual=True):
        loss_reduced = {}
        for key in losses_reduced[0].keys():
            val = [x[key].view(-1) for x in losses_reduced]
            if val[0].numel() == 2:
                # 新版: [loss_sum, num_tokens] 形式，跨 DP 聚合后取平均
                val = torch.vstack(val).sum(dim=0)
                torch.distributed.all_reduce(
                    val,
                    group=mpu.get_data_parallel_group(with_context_parallel=True)
                )
                loss_reduced[key] = val[0] / val[1]
            else:
                # 旧版: 直接取平均
                val = torch.cat(val).mean()
                loss_reduced[key] = val
    return loss_reduced, skipped_iter, ...
```

## 4. Forward-Backward 调度策略

**文件位置**: `megatron/core/pipeline_parallel/schedules.py`

### 4.1 调度策略选择

```python
def get_forward_backward_func(pp_size=None, vp_size=None):
    """根据并行配置自动选择调度策略"""
    pp_size = pp_size or parallel_state.get_pipeline_model_parallel_world_size()
    vp_size = vp_size or parallel_state.get_virtual_pipeline_model_parallel_world_size()

    if pp_size == 1:
        # 无 Pipeline Parallel: 简单的前向-反向
        return forward_backward_no_pipelining
    elif vp_size is not None and vp_size > 1:
        # Virtual Pipeline Parallel (交错调度)
        return forward_backward_pipelining_with_interleaving
    else:
        # 标准 1F1B 调度
        return forward_backward_pipelining_without_interleaving
```

### 4.2 无 PP 调度 (forward_backward_no_pipelining)

```
适用场景: PP size = 1

时间线:
GPU 0: F0 F1 F2 F3 B3 B2 B1 B0
       (4个 micro-batch 依次前向，再反向)

特点:
- 所有 micro-batch 顺序执行
- 梯度累积后统一更新
- 实现最简单
```

```python
def forward_backward_no_pipelining(
    forward_step_func,
    data_iterator,
    model,
    num_microbatches,
    ...
):
    losses_reduced = []
    for i in range(num_microbatches):
        # 前向传播
        output, loss_func = forward_step_func(data_iterator, model[0])

        # 反向传播
        if not forward_only:
            loss = loss_func(output)
            loss.backward()

        losses_reduced.append({'lm loss': loss.detach()})
    return losses_reduced
```

### 4.3 标准 1F1B 调度 (forward_backward_pipelining_without_interleaving)

```
适用场景: PP size > 1, 无 VP

时间线 (4 stages, 8 micro-batches):
Stage 0: F0 F1 F2 F3 B0 F4 B1 F5 B2 F6 B3 F7 B4 B5 B6 B7
Stage 1:    F0 F1 F2 B0 F3 B1 F4 B2 F5 B3 F6 B4 F7 B5 B6 B7
Stage 2:       F0 F1 B0 F2 B1 F3 B2 F4 B3 F5 B4 F6 B5 F7 B6 B7
Stage 3:          F0 B0 F1 B1 F2 B2 F3 B3 F4 B4 F5 B5 F6 B6 F7 B7

3个阶段:
1. Warmup: 只做 forward，填满 pipeline
2. Steady state (1F1B): 交替 forward 和 backward
3. Cooldown: 只做 backward，排空 pipeline
```

```python
# 1F1B 调度核心逻辑
num_warmup = min(pipeline_parallel_size, num_microbatches)
num_remaining = num_microbatches - num_warmup

input_tensors = []   # 保存前向激活值
output_tensors = []  # 保存前向输出

# Warmup 阶段
for i in range(num_warmup):
    input_tensor = recv_forward()           # 接收上一 stage 的激活
    output_tensor = forward_step(input_tensor)
    send_forward(output_tensor)             # 发送到下一 stage
    input_tensors.append(input_tensor)
    output_tensors.append(output_tensor)

# Steady state (1F1B)
for i in range(num_remaining):
    # Forward
    input_tensor = recv_forward()
    output_tensor = forward_step(input_tensor)
    send_forward(output_tensor)
    # Backward
    output_tensor = output_tensors.pop(0)
    input_tensor_grad = backward_step(output_tensor)
    send_backward(input_tensor_grad)
    recv_backward()

# Cooldown 阶段
for i in range(num_warmup):
    input_tensor = input_tensors.pop(0)
    output_tensor = output_tensors.pop(0)
    input_tensor_grad = backward_step(output_tensor)
    send_backward(input_tensor_grad)
```

**气泡比例 (Bubble Fraction)**:
```
Bubble = (pp_size - 1) / num_microbatches
```
- `pp_size=4, num_microbatches=8` → bubble = 37.5%
- 增加 `num_microbatches` 可减少气泡

### 4.4 交错 1F1B 调度 (Virtual Pipeline)

```
适用场景: PP size > 1, vp_size > 1

原理:
- 每个 GPU 存储多个 model chunk (vp_size 个)
- 每个 chunk 处理不同层范围
- 交错执行减少气泡

时间线 (2 stages, vp_size=2, 4 micro-batches):
Stage 0 (Chunks: 0,2): F0c0 F1c0 F0c1 F1c1 B0c1 B1c1 B0c0 B1c0
Stage 1 (Chunks: 1,3):   F0c0 F1c0 F0c1 F1c1 B0c1 B1c1 B0c0 B1c0

气泡比例:
Bubble = (pp_size - 1) / (vp_size * num_microbatches)
```

## 5. 用户接口: forward_step_func

### 5.1 标准实现模板

```python
def forward_step(data_iterator, model):
    """用户定义的前向步骤

    Args:
        data_iterator: 数据迭代器
        model: 当前 PP stage 的模型

    Returns:
        output_tensor: 前向输出 (传递给下一 stage 或计算 loss)
        loss_func: 损失函数 (只有 PP 最后一 stage 非 None)
    """
    # 获取当前 micro-batch 数据
    batch = get_batch(data_iterator)
    tokens = batch['tokens']
    labels = batch['labels']
    loss_mask = batch['loss_mask']

    # 模型前向
    output_tensor = model(
        input_ids=tokens,
        position_ids=batch['position_ids'],
        attention_mask=batch['attention_mask'],
    )

    # 返回输出和损失函数
    return output_tensor, partial(loss_func, labels, loss_mask)


def loss_func(labels, loss_mask, output_tensor):
    """计算损失

    Returns:
        loss: 标量损失值
        loss_dict: 用于日志记录的字典
    """
    losses = output_tensor.float()
    loss_mask = loss_mask.view(-1).float()

    # CrossEntropy Loss
    loss = torch.sum(losses.view(-1) * loss_mask) / loss_mask.sum()

    # 用于跨 DP ranks 聚合的格式
    # [loss_sum, num_tokens] 便于加权平均
    averaged_loss = average_losses_across_data_parallel_group([loss])
    return loss, {'lm loss': averaged_loss[0]}
```

## 6. 激活重计算 (Gradient Checkpointing)

### 6.1 配置选项

```python
@dataclass
class TransformerConfig:
    recompute_granularity: str = None
    """重计算粒度: 'full' | 'selective'"""

    recompute_method: str = None
    """重计算方法: 'uniform' | 'block'"""

    recompute_num_layers: int = None
    """'block' 方式下重计算的层数"""

    num_microbatches_with_partial_activation_checkpoints: int = None
    """部分激活检查点的 micro-batch 数"""
```

### 6.2 重计算模式

```
模式对比:

full (完全重计算):
  显存: 最小 (每层只保存输入)
  计算: 最大 (整层重算)
  适用: 显存极度受限

selective (选择性重计算):
  显存: 适中 (只保存昂贵操作的输出)
  计算: 较小 (只重算 Attention 等关键操作)
  适用: 平衡显存和计算

无重计算:
  显存: 最大 (保存所有激活)
  计算: 最小
  适用: 小模型或显存充足
```

```python
# 在 TransformerLayer 中
class TransformerLayer:
    def forward(self, hidden_states, ...):
        if self.config.recompute_granularity == 'full':
            # 使用 torch.utils.checkpoint 包装
            return self._checkpointed_forward(hidden_states, ...)
        else:
            return self._forward(hidden_states, ...)

    def _checkpointed_forward(self, *args):
        return checkpoint(self._forward, *args)
```

## 7. 训练日志 (training_log)

### 7.1 记录指标

```python
def training_log(loss_dict, total_loss_dict, learning_rate, iteration, ...):
    """记录训练指标

    记录内容:
    - lm loss (语言模型损失)
    - learning rate (当前学习率)
    - loss scale (FP16 损失缩放因子)
    - grad norm (梯度范数)
    - num zeros in grad (梯度中零的数量)
    - throughput (TFLOPS)
    - memory usage (显存使用)
    """
    # 计算吞吐量
    elapsed_time = timers('interval-time').elapsed(reset=True)
    throughput = num_floating_point_ops / elapsed_time / 1e12  # TFLOPS

    # 输出到控制台
    log_string = f'iteration {iteration}/{args.train_iters} | '
    log_string += f'consumed samples: {consumed_samples} | '
    log_string += f'elapsed time per iteration (ms): {elapsed_time * 1000.0 / args.log_interval:.1f} | '
    log_string += f'learning rate: {learning_rate:.3E} | '
    log_string += f'global batch size: {global_batch_size} | '
    log_string += f'lm loss: {loss:.6E} | '

    # 写入 TensorBoard / WandB
    if writer:
        writer.add_scalar('lm-loss-training/loss', loss, iteration)
        writer.add_scalar('learning-rate/learning-rate', learning_rate, iteration)
```

### 7.2 FLOPS 计算

```python
def num_floating_point_operations(args, batch_size):
    """估算每个 global batch 的浮点运算次数

    基于 Megatron 论文的分析:
    每个 transformer 层的 FLOPS ≈ 24 * seq_len * hidden_size^2

    加上:
    - Attention: 4 * seq_len^2 * hidden_size
    - MLP: 8 * seq_len * hidden_size * ffn_hidden_size
    """
    hidden_size = args.hidden_size
    seq_len = args.seq_length
    num_layers = args.num_layers

    # 简化公式 (不含激活重计算)
    flops_per_iter = (
        24 * batch_size * seq_len * num_layers * hidden_size * hidden_size
        + 4 * batch_size * num_layers * seq_len * seq_len * hidden_size
    )

    return flops_per_iter
```

## 9. 性能优化建议

### 9.1 Pipeline 效率

```python
# 推荐配置: num_microbatches >> pp_size
# 气泡比例 = (pp_size - 1) / num_microbatches

# 示例: pp=4, global_batch=2048, micro_batch=2, dp=8
num_microbatches = 2048 // (2 * 8) = 128
bubble = (4 - 1) / 128 = 2.3%  # 非常低的气泡率

# 使用 VP 进一步降低气泡
args.virtual_pipeline_model_parallel_size = 2
# 气泡 = (pp_size - 1) / (vp_size * num_microbatches)
#      = 3 / (2 * 128) = 1.2%
```

### 9.2 梯度累积与内存

```python
# 减少显存占用: 启用激活重计算
args.recompute_granularity = 'selective'  # 推荐: 性能/显存平衡
# args.recompute_granularity = 'full'    # 最节省显存

# 增大 micro-batch 利用率
# micro_batch_size 应尽量大到不 OOM 为止

# 梯度累积 = num_microbatches
# 大 num_microbatches → 等价于大 global batch size
# 但 PP 效率也更高
```

## 10. 常见问题

### Q1: Loss 不收敛 / NaN

**排查步骤**:
```python
# 1. 检查 loss scale
# 如果 loss scale 持续降低，说明梯度有 NaN/Inf
# 正常: loss scale 稳定在 65536 左右

# 2. 检查 grad norm
# 正常: grad norm 稳定在 0.1 ~ 10 之间
# 异常: grad norm 急剧增大

# 3. 减小学习率
args.lr = args.lr * 0.1

# 4. 减小初始化标准差
args.init_method_std = 0.01
```

### Q2: GPU 利用率低

**排查步骤**:
```python
# 1. 检查数据加载是否是瓶颈
# 使用 --profile 分析

# 2. 检查 PP 气泡
# 增大 num_microbatches

# 3. 检查通信是否重叠
args.overlap_grad_reduce = True
args.overlap_param_gather = True
```

### Q3: PP Stage 不平衡

**症状**: 某些 stage 比其他 stage 慢很多

**解决方案**:
```python
# 使用 PP split rank 不均等分配层
# 例如第一 stage 少放几层 (embedding 额外开销)
args.pipeline_model_parallel_split_rank = 1  # Encoder/Decoder 分界点

# 或使用均匀分配并配合 VP
args.virtual_pipeline_model_parallel_size = 2
```
