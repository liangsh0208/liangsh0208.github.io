## 1. 模块概述

Megatron-LM 支持多种并行策略，核心通信模块负责协调各并行维度间的数据传输。

```
┌─────────────────────────────────────────────────────────┐
│                    并行策略矩阵                          │
├─────────────────────────────────────────────────────────┤
│  Tensor Parallel (TP)    │ 模型层内切分                  │
│  Pipeline Parallel (PP)  │ 模型层间切分                  │
│  Data Parallel (DP)      │ 数据副本并行                  │
│  Sequence Parallel (SP)  │ 序列维度切分                  │
│  Expert Parallel (EP)    │ MoE 专家切分                  │
├─────────────────────────────────────────────────────────┤
│                    通信原语                              │
│  All-Reduce    │ 全局聚合求和                            │
│  All-Gather    │ 全局收集拼接                            │
│  Reduce-Scatter│ 分片聚合                                │
│  Scatter       │ 分发                                    │
│  Gather        │ 收集                                    │
│  Send/Recv     │ 点对点通信                              │
└─────────────────────────────────────────────────────────┘
```

## 2. Process Group 管理

**文件位置**: `megatron/core/parallel_state.py`

### 2.1 进程组初始化

```python
# megatron/core/parallel_state.py

def initialize_model_parallel(
    tensor_model_parallel_size: int = 1,
    pipeline_model_parallel_size: int = 1,
    virtual_pipeline_model_parallel_size: int = None,
    pipeline_model_parallel_split_rank: int = None,
    use_sharp: bool = False,
):
    """初始化模型并行进程组

    创建多组进程组用于不同并行维度:
    - _TENSOR_MODEL_PARALLEL_GROUP: TP 通信
    - _PIPELINE_MODEL_PARALLEL_GROUP: PP 通信
    - _DATA_PARALLEL_GROUP: DP 通信
    - _MODEL_PARALLEL_GROUP: TP+PP 全局
    """
    # 确保只初始化一次
    assert not _MODEL_PARALLEL_INITIALIZED, "Already initialized"

    world_size = torch.distributed.get_world_size()
    rank = torch.distributed.get_rank()

    # 计算各并行维度
    model_parallel_size = tensor_model_parallel_size * pipeline_model_parallel_size
    data_parallel_size = world_size // model_parallel_size

    # 验证配置
    assert world_size % model_parallel_size == 0

    num_tensor_model_parallel_groups = world_size // tensor_model_parallel_size
    num_pipeline_model_parallel_groups = world_size // pipeline_model_parallel_size
    num_data_parallel_groups = world_size // data_parallel_size

    # 构建各进程组
    for i in range(num_tensor_model_parallel_groups):
        start_rank = i * tensor_model_parallel_size
        end_rank = start_rank + tensor_model_parallel_size
        ranks = list(range(start_rank, end_rank))
        group = torch.distributed.new_group(ranks)
        if rank in ranks:
            _TENSOR_MODEL_PARALLEL_GROUP = group

    for i in range(num_pipeline_model_parallel_groups):
        ranks = [i + j * num_pipeline_model_parallel_groups
                 for j in range(pipeline_model_parallel_size)]
        group = torch.distributed.new_group(ranks)
        if rank in ranks:
            _PIPELINE_MODEL_PARALLEL_GROUP = group

    for i in range(num_data_parallel_groups):
        ranks = [i + j * num_data_parallel_groups
                 for j in range(data_parallel_size)]
        group = torch.distributed.new_group(ranks)
        if rank in ranks:
            _DATA_PARALLEL_GROUP = group
```

## 3. Tensor Parallel 通信

**文件位置**: `megatron/core/tensor_parallel/mappings.py`

### 3.1 通信原语实现

```python
# megatron/core/tensor_parallel/mappings.py

class _ReduceFromModelParallelRegion(torch.autograd.Function):
    """All-Reduce: TP 组内求和

    Forward: all-reduce
    Backward: 恒等 (梯度已在各 rank 独立计算)
    """
    @staticmethod
    def forward(ctx, input_):
        return _reduce(input_)

    @staticmethod
    def backward(ctx, grad_output):
        return grad_output  # 无需额外通信


class _ScatterToModelParallelRegion(torch.autograd.Function):
    """Scatter: 按 TP 切分

    Forward: scatter 到各 rank
    Backward: all-gather 聚合梯度
    """
    @staticmethod
    def forward(ctx, input_):
        return _split_along_last_dim(input_)

    @staticmethod
    def backward(ctx, grad_output):
        return _gather_along_last_dim(grad_output)


class _GatherFromModelParallelRegion(torch.autograd.Function):
    """Gather: 聚合 TP 分片

    Forward: all-gather 拼接
    Backward: scatter 分发梯度
    """
    @staticmethod
    def forward(ctx, input_):
        return _gather_along_last_dim(input_)

    @staticmethod
    def backward(ctx, grad_output):
        return _split_along_last_dim(grad_output)
```

### 3.2 维度操作函数

```python
def _split_along_last_dim(input_, group=None):
    """按最后一维切分

    输入: [batch, seq, hidden]
    输出: [batch, seq, hidden/tp_size]
    """
    world_size = get_tensor_model_parallel_world_size()
    input_list = torch.chunk(input_, world_size, dim=-1)
    rank = get_tensor_model_parallel_rank()
    output = input_list[rank].contiguous()
    return output


def _gather_along_last_dim(input_, group=None):
    """按最后一维聚合

    输入: [batch, seq, hidden/tp_size]
    输出: [batch, seq, hidden]
    """
    world_size = get_tensor_model_parallel_world_size()
    tensor_list = [torch.empty_like(input_) for _ in range(world_size)]
    torch.distributed.all_gather(tensor_list, input_, group=group)
    output = torch.cat(tensor_list, dim=-1).contiguous()
    return output


def _reduce(input_, group=None):
    """All-Reduce 求和"""
    torch.distributed.all_reduce(input_, group=group)
    return input_
```

### 3.3 TP 通信模式总结

| 操作 | Forward | Backward |
|------|---------|----------|
| `reduce_from_tensor_model_parallel_region` | All-Reduce | Identity |
| `scatter_to_tensor_model_parallel_region` | Scatter | All-Gather |
| `gather_from_tensor_model_parallel_region` | All-Gather | Scatter |

## 4. Pipeline Parallel 通信

**文件位置**: `megatron/core/pipeline_parallel/p2p_communication.py`

### 4.1 P2P 通信器

```python
# megatron/core/pipeline_parallel/p2p_communication.py

class P2PCommunicator:
    """Pipeline 并行点对点通信器

    处理相邻 PP stage 之间的张量传输
    """
    def __init__(self, config, pg_collection):
        self.config = config
        self.pp_group = pg_collection.pp

    def _communicate(self, send_tensor, recv_tensor, send_dst, recv_src):
        """双向通信: 同时发送和接收

        提升通信效率，避免等待
        """
        ops = []

        if send_tensor is not None:
            send_op = torch.distributed.P2POp(
                torch.distributed.isend,
                send_tensor,
                send_dst,
                group=self.pp_group
            )
            ops.append(send_op)

        if recv_tensor is not None:
            recv_op = torch.distributed.P2POp(
                torch.distributed.irecv,
                recv_tensor,
                recv_src,
                group=self.pp_group
            )
            ops.append(recv_op)

        # 批量执行通信
        reqs = torch.distributed.batch_isend_irecv(ops)
        for req in reqs:
            req.wait()

    def send_forward(self, tensor):
        """发送 forward 激活值到下一个 stage"""
        if not is_pipeline_last_stage():
            dst = get_pipeline_model_parallel_rank() + 1
            torch.distributed.send(tensor, dst, group=self.pp_group)

    def recv_forward(self, shape):
        """从上一个 stage 接收 forward 激活值"""
        if not is_pipeline_first_stage():
            src = get_pipeline_model_parallel_rank() - 1
            tensor = torch.empty(shape)
            torch.distributed.recv(tensor, src, group=self.pp_group)
            return tensor
        return None
```

### 4.2 通信调度模式

```python
# 1F1B (One Forward One Backward) 调度
def forward_backward_1f1b(
    forward_step_func,
    data_iterator,
    model,
    num_microbatches,
    forward_only=False,
):
    """1F1B 调度: 交替执行前向和反向

    优势:
    - 减少激活内存占用
    - 均衡各 stage 负载

    时间线:
    Stage 0: F0 F1 F2 F3 B0 F4 B1 F5 B2 F6 B3 ...
    Stage 1:    F0 F1 F2 B0 F3 B1 F4 B2 F5 B3 ...
    Stage 2:       F0 F1 B0 F2 B1 F3 B2 F4 B3 ...
    """
    num_warmup = min(pipeline_parallel_size, num_microbatches)
    num_remaining = num_microbatches - num_warmup

    input_tensors = []
    output_tensors = []

    # Warmup 阶段: 只做 forward
    for i in range(num_warmup):
        input_tensor = recv_forward()
        output_tensor = forward_step_func(input_tensor)
        send_forward(output_tensor)
        input_tensors.append(input_tensor)
        output_tensors.append(output_tensor)

    # Steady state: 1F1B
    for i in range(num_remaining):
        # Forward
        input_tensor = recv_forward()
        output_tensor = forward_step_func(input_tensor)
        send_forward(output_tensor)

        # Backward
        output_tensor = output_tensors.pop(0)
        input_tensor = input_tensors.pop(0)
        grad_input = backward_step_func(output_tensor)
        send_backward(grad_input)
        recv_backward()

    # Cooldown 阶段: 只做 backward
    for i in range(num_warmup):
        output_tensor = output_tensors.pop(0)
        input_tensor = input_tensors.pop(0)
        grad_input = backward_step_func(output_tensor)
        send_backward(grad_input)
        if i < num_warmup - 1:
            recv_backward()
```

## 5. Sequence Parallel 通信

**文件位置**: `megatron/core/tensor_parallel/mappings.py`

### 5.1 SP 通信原语

```python
def scatter_to_sequence_parallel_region(input_):
    """将序列按 TP rank 分片

    输入: [seq_len, batch, hidden]
    输出: [seq_len/tp_size, batch, hidden]

    用途: 在 Transformer Layer 开头分发序列
    """
    return _SplitAlongFirstDim.apply(input_)


def gather_from_sequence_parallel_region(input_):
    """聚合序列分片

    输入: [seq_len/tp_size, batch, hidden]
    输出: [seq_len, batch, hidden]

    用途: 在 Transformer Layer 结尾聚合序列
    """
    return _GatherAlongFirstDim.apply(input_)


def reduce_scatter_to_sequence_parallel_region(input_):
    """Reduce-Scatter 沿第一维

    输入: [seq_len, batch, hidden]
    输出: [seq_len/tp_size, batch, hidden]

    组合操作: All-Reduce + Scatter
    用于梯度同步后分片
    """
    return _ReduceScatterAlongFirstDim.apply(input_)
```

### 5.2 SP 在 Transformer 中的位置

```

├─────────────────────────────────────────────────────┤

│  Input: [seq_len, batch, hidden]                   │
│         │                                          │
```

## 6. Data Parallel 通信

### 6.1 梯度同步

```python
# megatron/core/distributed/param_and_grad_buffer.py

class ParamAndGradBuffer:
    """参数和梯度缓冲区管理

    功能:
    - 梯度分桶 (bucketization)
    - 异步 All-Reduce
    - 计算通信重叠
    """
    def __init__(self, config, param_dtype, grad_dtype, params):
        self.config = config
        self.params = params

        # 按大小分桶
        self.buckets = self._build_buckets(params)

    def all_reduce_gradients(self):
        """异步梯度 All-Reduce

        流程:
        1. 按桶准备梯度
        2. 异步启动 All-Reduce
        3. 与计算重叠
        """
        for bucket in self.buckets:
            # 等待梯度就绪
            self._wait_gradients_ready(bucket)

            # 异步 All-Reduce
            bucket.all_reduce_handle = torch.distributed.all_reduce(
                bucket.grad_buffer,
                async_op=True,
                group=self.dp_group
            )

    def finish_gradient_sync(self):
        """等待所有 All-Reduce 完成"""
        for bucket in self.buckets:
            if bucket.all_reduce_handle is not None:
                bucket.all_reduce_handle.wait()
                # 将梯度写回参数
                self._copy_gradients_back(bucket)
```

### 6.2 梯度分桶策略

```python
def _build_buckets(self, params):
    """梯度分桶

    优势:
    - 减少通信次数
    - 更好的通信效率
    - 计算通信重叠

    策略:
    - 按参数大小逆序排序
    - 桶大小接近预设值
    """
    params_per_bucket = self.config.params_per_bucket

    # 按 size 排序，大参数优先
    sorted_params = sorted(params, key=lambda p: p.numel(), reverse=True)

    buckets = []
    current_bucket = []
    current_size = 0

    for param in sorted_params:
        current_bucket.append(param)
        current_size += param.numel()

        if current_size >= params_per_bucket:
            buckets.append(Bucket(current_bucket))
            current_bucket = []
            current_size = 0

    if current_bucket:
        buckets.append(Bucket(current_bucket))

    return buckets
```

```python
# 使用 send_forward_recv_backward 同时发送和接收
def send_forward_recv_backward(output_tensor):
    """同时发送 forward 激活值并接收 backward 梯度"""
    ops = []

    # Forward send
    if not is_pipeline_last_stage():
        ops.append(torch.distributed.P2POp(
            torch.distributed.isend, output_tensor, next_stage))

    # Backward recv
    if not is_pipeline_first_stage():
        ops.append(torch.distributed.P2POp(
            torch.distributed.irecv, grad_buffer, prev_stage))

    reqs = torch.distributed.batch_isend_irecv(ops)
    return reqs

def async_grad_reduce(grad_buffer):
```

### 8.3 常见问题

| 问题 | 症状 | 解决方案 |
|------|------|---------|
| TP 通信瓶颈 | GPU 利用率低 | 减小 TP size 或启用 SP |
| PP 气泡大 | 训练效率低 | 使用 VP 或增加 micro-batch |
| DP 同步慢 | 所有 GPU 等待 | 检查网络或减小 batch size |
| NCCL 超时 | 训练中断 | 增加 NCCL_TIMEOUT 环境变量 |
