### 2.2 参数分组键 (ParamKey)

```python
@dataclass(frozen=True, slots=True)
class ParamKey:
    """参数分组键

    用于将参数分组到不同的优化器配置
    """

    name: Union[str, Tuple[str]] = field(default_factory=tuple)
    """参数名称模式（支持通配符）"""

    attr: Union[str, Tuple[str]] = field(default_factory=tuple)
    """参数属性匹配"""

    predicate: Union[ParamPredicate, Tuple[ParamPredicate]] = field(default_factory=tuple)
    """谓词匹配"""

    def matches(self, param: torch.nn.Parameter, param_name: str) -> bool:
        """检查参数是否匹配此键"""
        # 检查名称匹配
        for target_name in self.name if isinstance(self.name, tuple) else [self.name]:
            if fnmatch.fnmatch(param_name, target_name):
                return True

        # 检查属性匹配
        for target_attr in self.attr if isinstance(self.attr, tuple) else [self.attr]:
            if getattr(param, target_attr, False):
                return True

        # 检查谓词匹配
        for pred in self.predicate if isinstance(self.predicate, tuple) else [self.predicate]:
            if pred(param):
                return True

        return False
```

### 3.2 学习率计算

```python
def get_lr(self, param_group: dict) -> float:
    """计算当前步的学习率

    调度流程:
    1. Warmup 阶段：线性增加
    2. 稳定阶段 或 衰减阶段
    """

    max_lr = param_group.get('max_lr', self.max_lr)
    min_lr = param_group.get('min_lr', self.min_lr)

    # 1. Warmup 阶段：线性增加
    if self.lr_warmup_steps > 0 and self.num_steps <= self.lr_warmup_steps:
        return self.init_lr + (
            (max_lr - self.init_lr) * float(self.num_steps) / float(self.lr_warmup_steps)
        )

    # 2. Constant 调度
    if self.lr_decay_style == 'constant':
        return max_lr

    # 3. 超过衰减步数，使用最小学习率
    if self.num_steps > self.lr_decay_steps:
        return min_lr

    # 4. Inverse Square Root 调度
    if self.lr_decay_style == 'inverse-square-root':
        warmup_steps = max(self.lr_warmup_steps, 1)
        num_steps = max(self.num_steps, 1)
        lr = max_lr * warmup_steps**0.5 / (num_steps**0.5)
        return max(min_lr, lr)

    # 5. 计算衰减比例
    num_steps_ = self.num_steps - self.lr_warmup_steps
    decay_steps_ = self.lr_decay_steps - self.lr_warmup_steps
    decay_ratio = float(num_steps_) / float(decay_steps_)
    delta_lr = max_lr - min_lr

    # 6. 线性衰减
    if self.lr_decay_style == 'linear':
        coeff = 1.0 - decay_ratio

    # 7. Cosine 衰减
    elif self.lr_decay_style == 'cosine':
        coeff = 0.5 * (math.cos(math.pi * decay_ratio) + 1.0)

    # 8. WSD (Warmup-Stable-Decay)
    elif self.lr_decay_style == 'WSD':
        wsd_anneal_start_ = self.lr_decay_steps - self.wsd_decay_steps
        if self.num_steps <= wsd_anneal_start_:
            coeff = 1.0  # 稳定阶段
        else:
            # 衰减阶段
            wsd_steps = self.num_steps - wsd_anneal_start_
            wsd_decay_ratio = float(wsd_steps) / float(self.wsd_decay_steps)

            if self.lr_wsd_decay_style == "linear":
                coeff = 1.0 - wsd_decay_ratio
            elif self.lr_wsd_decay_style == "cosine":
                coeff = 0.5 * (math.cos(math.pi * wsd_decay_ratio) + 1.0)
            elif self.lr_wsd_decay_style == "exponential":
                coeff = (2.0 * math.pow(0.5, wsd_decay_ratio)) - 1.0
            elif self.lr_wsd_decay_style == "minus_sqrt":
                coeff = 1.0 - math.sqrt(wsd_decay_ratio)

    return min_lr + coeff * delta_lr
```

### 3.3 学习率调度曲线

```
学习率
    │
max │      ╭──────────────────────╮
    │     /                        \
    │    /                          \
    │   /                            \
    │  /                              \
min │─╯                                ╰────────
    └─────────────────────────────────────────→ 训练步数
      │warmup│    stable    │  decay  │
```

**WSD 调度特点**:
- Warmup: 线性增长
- Stable: 保持最高学习率
- Decay: 快速衰减（支持多种衰减样式）

## 4. 权重衰减调度

```python
def get_wd(self, param_group: Optional[dict] = None) -> float:
    """计算当前步的权重衰减

    支持 linear 和 cosine 增量
    """
    start_wd = param_group.get('start_wd', self.start_wd) if param_group else self.start_wd
    end_wd = param_group.get('end_wd', self.end_wd) if param_group else self.end_wd

    # 超过增量步数，使用最终值
    if self.num_steps > self.wd_incr_steps:
        return end_wd

    # 常量模式
    if self.wd_incr_style == 'constant':
        assert start_wd == end_wd
        return end_wd

    incr_ratio = float(self.num_steps) / float(self.wd_incr_steps)
    delta_wd = end_wd - start_wd

    # 线性增量
    if self.wd_incr_style == 'linear':
        coeff = incr_ratio
    # Cosine 增量
    elif self.wd_incr_style == 'cosine':
        coeff = 0.5 * (math.cos(math.pi * (1 - incr_ratio)) + 1.0)

    return start_wd + coeff * delta_wd
```

### 5.2 梯度裁剪

```python
def clip_grad_norm(self, clip_grad: float) -> float:
    """梯度裁剪

    基于 L2 范数裁剪梯度，确保梯度更新稳定

    Args:
        clip_grad: 裁剪阈值

    Returns:
        总梯度范数
    """
    params = []
    for param_group in self.optimizer.param_groups:
        for param in param_group['params']:
            if param.grad is not None:
                params.append(param.grad)

    # 计算总梯度范数
    total_norm = 0.0
    for grad in params:
        grad_norm = grad.data.norm(2)
        total_norm += grad_norm ** 2
    total_norm = total_norm ** 0.5

    # 跨 rank 同步梯度范数
    total_norm_cuda = torch.cuda.FloatTensor([float(total_norm)])
    torch.distributed.all_reduce(
        total_norm_cuda,
        op=torch.distributed.ReduceOp.MAX,
        group=self.model_parallel_group
    )
    total_norm = total_norm_cuda[0].item()

    # 裁剪
    if clip_grad > 0.0:
        clip_coeff = clip_grad / (total_norm + 1e-6)
        if clip_coeff < 1.0:
            for grad in params:
                grad.data.mul_(clip_coeff)

    return total_norm
```

### 5.3 损失缩放 (Loss Scaling)

```python
class MegatronGradScaler:
    """梯度缩放器

    用于 FP16 训练时防止梯度下溢
    """

    def __init__(
        self,
        initial_scale: float = 2**32,
        min_scale: float = 1.0,
        growth_factor: float = 2.0,
        backoff_factor: float = 0.5,
        growth_interval: int = 2000,
        hysteresis: int = 2,
    ):
        self._scale = initial_scale
        self._min_scale = min_scale
        self._growth_factor = growth_factor
        self._backoff_factor = backoff_factor
        self._growth_interval = growth_interval
        self._hysteresis = hysteresis

        self._growth_tracker = 0
        self._found_inf_or_nan = False
        self._hysteresis_tracker = 0

    def scale(self, loss: torch.Tensor) -> torch.Tensor:
        """缩放损失"""
        return loss * self._scale

    def unscale_(self, optimizer: torch.optim.Optimizer):
        """反缩放梯度"""
        inv_scale = 1.0 / self._scale
        for param_group in optimizer.param_groups:
            for param in param_group['params']:
                if param.grad is not None:
                    param.grad.data.mul_(inv_scale)

    def step(self, optimizer: torch.optim.Optimizer):
        """优化器步骤（带梯度检查）"""
        # 检查是否溢出
        self._found_inf_or_nan = self._check_for_inf_or_nan(optimizer)

        if not self._found_inf_or_nan:
            optimizer.step()
            self._growth_tracker += 1

            # 定期增加 scale
            if self._growth_tracker >= self._growth_interval:
                self._scale *= self._growth_factor
                self._growth_tracker = 0
        else:
            # 溢出时减小 scale
            self._scale *= self._backoff_factor
            self._scale = max(self._scale, self._min_scale)
            self._growth_tracker = 0

            optimizer.zero_grad()
```

## 6. 分布式优化器 (DistributedOptimizer)

**文件位置**: `megatron/core/optimizer/distrib_optimizer.py`

### 6.1 核心思想

分布式优化器将优化器状态分片存储在不同的 DP rank 上：

```
传统方式 (每个 rank 存储完整状态):
Rank 0: [Param0, Param1, Param2, Param3] + [State0, State1, State2, State3]
Rank 1: [Param0, Param1, Param2, Param3] + [State0, State1, State2, State3]

分布式优化器 (状态分片):
Rank 0: [Param0, Param1, Param2, Param3] + [State0]
Rank 1: [Param0, Param1, Param2, Param3] + [State1]
```

### 6.2 Range 类

```python
class Range:
    """范围索引类

    表示参数分片的范围
    """
    def __init__(self, start: int, end: int):
        self.start = start
        self.end = end
        self.size = end - start

    def normalize(self, start: int = 0):
        """移动范围的起始位置"""
        return Range(start, start + self.size)
```

### 6.3 参数分片映射

```python
@classmethod
def _build_model_gbuf_param_range_map(
    cls,
    param_world_index_map: Dict[torch.nn.Parameter, Tuple],
    gbuf_world_range: Range,
    bucket_offset: int,
):
    """构建参数到梯度缓冲区分片范围的映射

    每个 DP rank '拥有' 一段连续的梯度缓冲区区域
    """
    param_range_map = {}
    for param, param_world_indexes in param_world_index_map.items():
        param_world_start, param_world_end, _ = param_world_indexes

        # 计算参数在本 rank 缓冲区内的范围
        param_local_start = max(0, param_world_start - gbuf_world_range.start)
        param_local_end = min(gbuf_world_range.size, param_world_end - gbuf_world_range.start)

        if param_local_end > param_local_start:
            param_range_map[param] = {
                "gbuf_world": Range(param_local_start + gbuf_world_range.start,
                                    param_local_end + gbuf_world_range.start),
                "gbuf_local": Range(param_local_start, param_local_end),
                "param": Range(max(0, gbuf_world_range.start - param_world_start),
                               max(0, gbuf_world_range.start - param_world_start) + (param_local_end - param_local_start)),
            }

    return param_range_map
```

### 6.4 步骤流程

```python
def step(self, args):
    """分布式优化器更新步骤

    流程:
    1. Reduce-Scatter 梯度
    2. 本地优化器更新
    3. All-Gather 参数
    """
    # 1. 同步梯度缓冲区
    for bucket in self.buckets:
        # Reduce-scatter: 每个 rank 获取一部分梯度
        torch.distributed.reduce_scatter_tensor(
            bucket.gbuf_reduced,
            bucket.gbuf,
            op=torch.distributed.ReduceOp.SUM,
            group=self.data_parallel_group,
        )

    # 2. 优化器步骤 (只更新本 rank 负责的参数)
    self.optimizer.step()

    # 3. All-gather: 同步更新后的参数
    for bucket in self.buckets:
        torch.distributed.all_gather_into_tensor(
            bucket.param,
            bucket.param_shard,
            group=self.data_parallel_group,
        )
```

## 7. 混合设备优化器 (HybridDeviceOptimizer)

**文件位置**: `megatron/core/optimizer/cpu_offloading/hybrid_optimizer.py`

### 7.1 用途

将部分优化器状态卸载到 CPU，减少 GPU 内存占用。

```python
class HybridDeviceOptimizer(torch.optim.Optimizer):
    """混合设备优化器

    将部分参数在 GPU 上更新，部分在 CPU 上更新
    """

    def __init__(
        self,
        params,
        offload_fraction=0.5,     # 卸载比例
        cpu_optimizer_cls=None,   # CPU 优化器类
        gpu_optimizer_cls=None,   # GPU 优化器类
        param_update_in_fp32: bool = False,
        pin_cpu_grads: bool = True,
        pin_cpu_params: bool = True,
        overlap_cpu_optimizer_d2h_h2d: bool = True,  # 重叠数据传输
        **kwargs,
    ):
        ...
```

### 7.2 工作流程

```python
def step(self, closure=None):
    """更新步骤

    流程:
    1. 同步参数组到子优化器
    2. GPU -> CPU 梯度传输
    3. 执行优化器步骤
    4. 同步状态回主优化器
    """
    # 同步参数组
    self._sync_hdo_param_groups_to_sub_optimizers()

    # 异步梯度传输 D2H
    self._d2h_stream.wait_stream(torch.cuda.current_stream())
    with torch.cuda.stream(self._d2h_stream):
        self._set_sub_optimizer_grads()

    # GPU 优化器步骤
    if self.gpu_optimizer:
        self.gpu_optimizer.step(closure)

    # CPU 优化器步骤 (等待 D2H 完成)
    for cpu_optimizer in self.cpu_optimizers:
        d2h_event = self._cpu_optimizer_map_data_event.pop(cpu_optimizer, None)
        if d2h_event is not None:
            d2h_event.synchronize()
        cpu_optimizer.step(closure)

    # 同步状态
    self._sync_sub_optimizers_state_to_hdo()
```

## 10. 常见问题

### Q1: Loss 变成 NaN

**可能原因**:
- 梯度爆炸
- Loss scale 过大
- 精度溢出

**解决方案**:
```python
# 1. 减小学习率
optimizer_config.lr = 5e-5

# 2. 使用更小的初始 loss scale
optimizer_config.initial_loss_scale = 2**16

# 3. 减小梯度裁剪阈值
optimizer_config.clip_grad = 0.5
```

### Q2: 训练不稳定

**可能原因**:
- 学习率过大
- 热身不足
- 权重衰减过大

**解决方案**:
```python
# 增加热身步数
lr_scheduler.lr_warmup_steps = 5000

# 减小权重衰减
optimizer_config.weight_decay = 0.001

# 使用 WSD 调度保持稳定
lr_scheduler.lr_decay_style = 'WSD'
```

### Q3: 内存不足

**可能原因**:
- 优化器状态过大

**解决方案**:
```python
# 1. 使用分布式优化器
optimizer_config.use_distributed_optimizer = True

# 2. CPU 卸载
optimizer_config.optimizer_cpu_offload = True
optimizer_config.optimizer_offload_fraction = 0.7

# 3. 使用精度感知优化器
optimizer_config.use_precision_aware_optimizer = True
optimizer_config.main_params_dtype = torch.bfloat16
```
