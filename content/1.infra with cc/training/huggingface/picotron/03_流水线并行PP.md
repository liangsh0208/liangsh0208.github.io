# 03_流水线并行PP

## 1. 目标论证

**为什么需要 Pipeline Parallel？**

随着模型深度增加(70B+模型通常有80层)，即使TP切分后单层可放下，整体模型参数仍超出显存。
- 问题: Llama-70B有80层transformer，每层约800M参数，总计64B
- 目标: 将层分配到不同设备，形成流水线阶段

**气泡问题：**
- 朴素流水线存在大量空闲时间(气泡)
- 需要调度策略最大化GPU利用率

## 2. 工作假设

**分段约束：**
- `num_layers` 可能在PP组间不均匀分配
- 各阶段计算时间平衡假设(层同质)

**激活检查点：**
- PP边界必须保存激活值用于反向传播
- 中间层可选择重计算节省显存

## 3. 解决方案

### 3.1 层分配算法

```python
def distribute_layers(num_layers, pp_rank, pp_world_size):
    """
    处理layers不能被pp_world_size整除的情况
    返回当前PP rank负责的层范围
    """
    layers_per_gpu = [
        num_layers // pp_world_size + 
        (1 if i < num_layers % pp_world_size else 0)
        for i in range(pp_world_size)
    ]
    start_layer = sum(layers_per_gpu[:pp_rank])
    return range(start_layer, start_layer + layers_per_gpu[pp_rank])
```

### 3.2 Pipeline Parallel容器

```python
class PipelineParallel(nn.Module):
    def __init__(self, model, config):
        super().__init__()
        pgm = get_process_group_manager()
        
        # 确定本进程负责的层
        self.layer_range = distribute_layers(
            config.num_hidden_layers,
            pgm.pp_rank, 
            pgm.pp_world_size
        )
        
        # 选择对应层的参数
        self.layers = nn.ModuleList([
            model.model.layers[i] for i in self.layer_range
        ])
        
        # 首/尾stage特殊处理
        self.is_first_stage = (pgm.pp_rank == 0)
        self.is_last_stage = (pgm.pp_rank == pgm.pp_world_size - 1)
        
        if self.is_first_stage:
            self.embed_tokens = model.model.embed_tokens
        if self.is_last_stage:
            self.norm = model.model.norm
            self.lm_head = model.lm_head
```

### 3.3 AFAB调度(All Forward All Backward)

```python
def train_step_pipeline_afab(model, data_loader, device):
    """
    简单调度: 先全部前向，再全部反向
    优点: 实现简单，激活管理清晰
    缺点: 气泡大(ND-1个空转周期)
    """
    pgm = get_process_group_manager()
    num_microbatches = get_grad_acc_steps()
    
    # Phase 1: Warmup - 填充流水线
    input_tensors = []
    for i in range(num_microbatches):
        # 接收上一stage的激活
        input_tensor = pipeline_communicate(
            operation='recv_forward',
            tensor_shape=..., device=device
        )
        input_tensors.append(input_tensor)
        
        # 本地前向计算
        output_tensor = model.forward(input_tensor)
        
        # 发送给下一stage
        pipeline_communicate(
            operation='send_forward',
            tensor=output_tensor
        )
    
    # Phase 2: Cooldown - 全部反向
    output_tensors_grad = []
    for i in range(num_microbatches):
        # 接收下一stage的梯度
        output_tensor_grad = pipeline_communicate(
            operation='recv_backward',
            tensor_shape=..., device=device
        )
        output_tensors_grad.append(output_tensor_grad)
        
        # 反向传播
        input_tensor_grad = model.backward(
            output_tensors_grad[-1], 
            input_tensors[-(i+1)]
        )
        
        # 发送给上一stage
        pipeline_communicate(
            operation='send_backward',
            tensor=input_tensor_grad
        )
```

### 3.4 1F1B调度(One Forward One Backward)

```python
def train_step_pipeline_1f1b(model, data_loader, device):
    """
    交错调度: 前向后立即反向
    优点: 气泡最小(1个周期)
    缺点: 需要同时持有前向和反向激活
    """
    pgm = get_process_group_manager()
    num_microbatches = get_grad_acc_steps()
    pp_rank, pp_world_size = pgm.pp_rank, pgm.pp_world_size
    
    # 计算warmup需要的前向次数
    num_warmup_microbatches = min(
        pp_world_size - pp_rank - 1,
        num_microbatches
    )
    
    # Phase 1: Warmup
    input_tensors = []
    for i in range(num_warmup_microbatches):
        input_tensor = recv_forward(...)
        input_tensors.append(input_tensor)
        
        output_tensor = model.forward(input_tensor)
        send_forward(output_tensor)
    
    # Phase 2: Steady State
    num_microbatches_remaining = num_microbatches - num_warmup_microbatches
    for i in range(num_microbatches_remaining):
        # 前向
        input_tensor = recv_forward(...) if not model.is_first_stage else get_microbatch()
        input_tensors.append(input_tensor)
        output_tensor = model.forward(input_tensor)
        
        # 双向通信: 发送激活 + 接收梯度
        output_tensor_grad = bidirectional_pipeline_communicate(
            operation='send_fwd_recv_bwd',
            send_tensor=output_tensor,
            recv_shape=...,
            device=device
        )
        
        # 反向
        input_tensor_grad = model.backward(
            output_tensor_grad,
            input_tensors.pop(0)  # 释放已处理的激活
        )
        
        # 双向通信: 发送梯度 + 接收激活
        input_tensor = bidirectional_pipeline_communicate(
            operation='send_bwd_recv_fwd',
            send_tensor=input_tensor_grad,
            recv_shape=...,
            device=device
        )
        input_tensors.append(input_tensor)
    
    # Phase 3: Cooldown
    for i in range(num_warmup_microbatches):
        output_tensor_grad = recv_backward(...)
        input_tensor_grad = model.backward(
            output_tensor_grad,
            input_tensors.pop(0)
        )
        send_backward(input_tensor_grad)
```

### 3.5 P2P通信实现

```python
def pipeline_communicate(operation, tensor=None, tensor_shape=None, device=None):
    """
    流水线stage间的点对点通信
    """
    pgm = get_process_group_manager()
    
    if operation == 'recv_forward':
        if pgm.pp_prev_rank is None:
            return None
        tensor = torch.empty(tensor_shape, device=device)
        dist.recv(tensor, src=pgm.pp_prev_rank)
        return tensor
    
    elif operation == 'send_forward':
        if pgm.pp_next_rank is None:
            return
        dist.send(tensor, dst=pgm.pp_next_rank)
    
    elif operation == 'recv_backward':
        if pgm.pp_next_rank is None:
            return None
        tensor = torch.empty(tensor_shape, device=device)
        dist.recv(tensor, src=pgm.pp_next_rank)
        return tensor
    
    elif operation == 'send_backward':
        if pgm.pp_prev_rank is None:
            return
        dist.send(tensor, dst=pgm.pp_prev_rank)

def bidirectional_pipeline_communicate(operation, send_tensor, recv_shape, device):
    """同时发送和接收，减少等待时间"""
    pgm = get_process_group_manager()
    recv_tensor = torch.empty(recv_shape, device=device)
    
    if operation == 'send_fwd_recv_bwd':
        # 发送激活给下一stage，同时接收来自下一stage的梯度
        dist.batch_isend_irecv([
            dist.P2POp(dist.isend, send_tensor, pgm.pp_next_rank),
            dist.P2POp(dist.irecv, recv_tensor, pgm.pp_next_rank)
        ])
    elif operation == 'send_bwd_recv_fwd':
        # 发送梯度给上一stage，同时接收来自上一stage的激活
        dist.batch_isend_irecv([
            dist.P2POp(dist.isend, send_tensor, pgm.pp_prev_rank),
            dist.P2POp(dist.irecv, recv_tensor, pgm.pp_prev_rank)
        ])
    
    return recv_tensor
```

## 4. 故障模式

| 问题 | 可能原因 | 解决方案 |
|-----|---------|---------|
| `RuntimeError: Connection reset` | PP相邻rank未匹配send/recv | 确保所有stage调用对称的通信 |
| Stage死锁 | 1F1B warmup计算错误 | 验证 `min(pp_size - rank - 1, microbatches)` |
| 激活尺寸不匹配 | 序列长度不能被CP整除 | 检查 `seq_len % (cp_size * pp_size)` |
| 梯度不收敛 | PP边界未设置requires_grad | 确保input_tensor保留计算图 |
| 内存溢出 | 1F1B保存过多激活 | 限制num_warmup_microbatches或使用激活检查点 |
