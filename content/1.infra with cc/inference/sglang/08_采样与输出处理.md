### 2.2 参数校验

```python
def verify(self, vocab_size: int):
    """参数合法性校验"""

    # 温度范围
    if self.temperature < 0:
        raise ValueError(f"temperature must be non-negative, got {self.temperature}")

    # Top-p 范围
    if not 0 < self.top_p <= 1:
        raise ValueError(f"top_p must be in (0, 1], got {self.top_p}")

    # Top-k 范围
    if self.top_k < -1 or self.top_k == 0:
        raise ValueError(f"top_k must be -1 or >= 1, got {self.top_k}")

    # 惩罚参数范围
    if not -2 <= self.frequency_penalty <= 2:
        raise ValueError(f"frequency_penalty must be in [-2, 2]")

    if not -2 <= self.presence_penalty <= 2:
        raise ValueError(f"presence_penalty must be in [-2, 2]")

    if self.repetition_penalty <= 0:
        raise ValueError(f"repetition_penalty must be positive")

    # 约束解码互斥检查
    constraint_count = sum([
        self.json_schema is not None,
        self.regex is not None,
        self.ebnf is not None,
    ])
    if constraint_count > 1:
        raise ValueError("Only one of json_schema, regex, ebnf can be set")
```

## 三、采样实现

### 3.1 Sampler 类

```python
class Sampler:
    """Token 采样器"""

    def __init__(self, model_config: ModelConfig):
        self.vocab_size = model_config.vocab_size

    def forward(
        self,
        logits: torch.Tensor,              # [batch_size, vocab_size]
        sampling_info: SamplingBatchInfo,
    ) -> torch.Tensor:
        """执行采样"""

        # 1. 应用 logit 偏置
        if sampling_info.logit_bias is not None:
            logits = logits + sampling_info.logit_bias

        # 2. 应用惩罚 (由 PenalizerOrchestrator 维护)
        if sampling_info.penalizer_orchestrator.is_required:
            logits = sampling_info.penalizer_orchestrator.apply(logits)

        # 3. 应用语法约束掩码
        if sampling_info.grammar_mask is not None:
            logits = logits.masked_fill(sampling_info.grammar_mask, float('-inf'))

        # 4. 采样
        if sampling_info.temperature == 0:
            # 贪婪采样
            sampled_ids = torch.argmax(logits, dim=-1)
        else:
            # 概率采样
            probs = torch.softmax(logits / sampling_info.temperature, dim=-1)

            # Top-k 过滤
            if sampling_info.top_k > 0:
                probs = self._top_k_filter(probs, sampling_info.top_k)

            # Top-p 过滤
            if sampling_info.top_p < 1.0:
                probs = self._top_p_filter(probs, sampling_info.top_p)

            # Min-p 过滤
            if sampling_info.min_p > 0:
                probs = self._min_p_filter(probs, sampling_info.min_p)

            # 采样
            sampled_ids = torch.multinomial(probs, num_samples=1).squeeze(-1)

        return sampled_ids
```

### 3.2 PenalizerOrchestrator 详解

**文件位置**：`python/sglang/srt/sampling/`

`PenalizerOrchestrator` 是处理**频率惩罚**、**存在惩罚**和**重复惩罚**的核心组件，它在每个 decode step 中高效维护每个请求的 token 出现统计，避免每次从完整输出序列重新计数：

```python
class PenalizerOrchestrator:
    """惩罚器的编排器，统一管理多种惩罚"""

    def __init__(self, vocab_size: int, device: str):
        self.vocab_size = vocab_size
        self.device = device
        self.is_required = False          # 如果所有请求都不需要惩罚，则跳过
        self.freq_penalties = None        # [batch_size]
        self.pres_penalties = None        # [batch_size]
        self.rep_penalties = None         # [batch_size]
        # 累积输出 token 计数
        self.cumulated_output_tokens = None  # List[Dict[int, int]]

    def cumulate_output_tokens(self, output_ids: torch.Tensor):
        """在每个 decode step 后，将新采样的 token 加入累积计数"""
        # output_ids: [batch_size]
        # 更新每个请求的出现次数
        for i, token_id in enumerate(output_ids.cpu().tolist()):
            self.cumulated_output_tokens[i][token_id] += 1

    def apply(self, logits: torch.Tensor) -> torch.Tensor:
        """将累积的惩罚应用到 logits 上"""
        batch_size = logits.shape[0]
        for i in range(batch_size):
            token_counts = self.cumulated_output_tokens[i]
            for token_id, count in token_counts.items():
                # 重复惩罚: 对大于0的logits做除法，小于0的做乘法
                if self.rep_penalties[i] != 1.0:
                    if logits[i, token_id] > 0:
                        logits[i, token_id] /= self.rep_penalties[i]
                    else:
                        logits[i, token_id] *= self.rep_penalties[i]
                # 频率惩罚: 按出现次数线性递减
                if self.freq_penalties[i] != 0:
                    logits[i, token_id] -= count * self.freq_penalties[i]
                # 存在惩罚: 只要出现过就递减一次
                if self.pres_penalties[i] != 0:
                    logits[i, token_id] -= self.pres_penalties[i]
        return logits
```

**关键设计**：
1. **惰性初始化**：如果 batch 中所有请求的 `frequency_penalty`、`presence_penalty`、`repetition_penalty` 均为默认值，`is_required=False`，整个惩罚逻辑被跳过
2. **累积而非重算**：不是每次从 `output_ids` 列表重新统计，而是每步只增量更新新 token 的计数，复杂度 O(batch_size) 而非 O(seq_len)
3. **与采样后端集成**：当使用 `flashinfer` 等 GPU 采样后端时，惩罚数据会随 `SamplingBatchInfo` 一起传入，由后端内核在 GPU 上直接应用，避免 CPU-GPU 数据拷贝

### 3.2 惩罚应用

```python
def _apply_penalties(
    self,
    logits: torch.Tensor,
    sampling_info: SamplingBatchInfo,
) -> torch.Tensor:
    """应用重复惩罚"""

    freq_penalty = sampling_info.frequency_penalty
    pres_penalty = sampling_info.presence_penalty
    rep_penalty = sampling_info.repetition_penalty

    # 获取已生成的 token 统计
    output_token_counts = sampling_info.output_token_counts

    for i in range(logits.shape[0]):
        for token_id, count in output_token_counts[i].items():
            # 重复惩罚
            if rep_penalty != 1.0:
                if logits[i, token_id] > 0:
                    logits[i, token_id] /= rep_penalty
                else:
                    logits[i, token_id] *= rep_penalty

            # 频率惩罚
            if freq_penalty != 0:
                logits[i, token_id] -= count * freq_penalty

            # 存在惩罚
            if pres_penalty != 0:
                logits[i, token_id] -= pres_penalty

    return logits
```

### 3.3 Top-k/Top-p/Min-p 过滤

```python
def _top_k_filter(self, probs: torch.Tensor, k: int) -> torch.Tensor:
    """Top-k 过滤：只保留概率最高的 k 个 token"""
    top_k_probs, top_k_indices = probs.topk(k, dim=-1)
    probs_filtered = torch.zeros_like(probs)
    probs_filtered.scatter_(dim=-1, index=top_k_indices, src=top_k_probs)
    return probs_filtered / probs_filtered.sum(dim=-1, keepdim=True)


def _top_p_filter(self, probs: torch.Tensor, p: float) -> torch.Tensor:
    """Top-p (Nucleus) 过滤：概率累计达到 p 的最小集合"""
    sorted_probs, sorted_indices = probs.sort(dim=-1, descending=True)
    cumsum_probs = sorted_probs.cumsum(dim=-1)

    # 找到超过 p 的位置
    cutoff_mask = cumsum_probs > p
    # 保留第一个超过 p 的位置
    cutoff_mask[:, 1:] = cutoff_mask[:, :-1].clone()
    cutoff_mask[:, 0] = False

    sorted_probs[cutoff_mask] = 0
    probs_filtered = probs.clone()
    probs_filtered.scatter_(dim=-1, index=sorted_indices, src=sorted_probs)
    return probs_filtered / probs_filtered.sum(dim=-1, keepdim=True)


def _min_p_filter(self, probs: torch.Tensor, min_p: float) -> torch.Tensor:
    """Min-p 过滤：只保留概率 >= min_p * max_prob 的 token"""
    max_probs = probs.max(dim=-1, keepdim=True)[0]
    min_threshold = min_p * max_probs
    mask = probs < min_threshold
    probs_filtered = probs.masked_fill(mask, 0)
    return probs_filtered / probs_filtered.sum(dim=-1, keepdim=True)
```

```python
        """处理请求的语法约束"""

        # 确定语法类型
        if req.sampling_params.json_schema:
            key = ("json", req.sampling_params.json_schema)
        elif req.sampling_params.regex:
            key = ("regex", req.sampling_params.regex)
        elif req.sampling_params.ebnf:
            key = ("ebnf", req.sampling_params.ebnf)
        elif req.sampling_params.structural_tag:
            key = ("structural_tag", req.sampling_params.structural_tag)
        else:
            return False

        # 获取编译后的语法 (可能异步)
        grammar_value, cache_hit = self.grammar_backend.get_cached_or_future_value(key)

        if grammar_value is None:
            # 需要等待编译
            req.grammar = None
            self.grammar_queue.append(req)
            return True
        else:
            req.grammar = grammar_value
            return False
```

```python
    def __init__(self, vocab_size: int):
        self.vocab_size = vocab_size
        self.cache: Dict[Tuple, BaseGrammarObject] = {}
        self.executor = ThreadPoolExecutor(max_workers=4)
```

```python
# 在采样时应用语法掩码
def sample_with_grammar(
    logits: torch.Tensor,
    grammar: BaseGrammarObject,
) -> int:
    """带语法约束的采样"""

    # 获取掩码
    bitmask = grammar.get_bitmask()  # [vocab_size], True = 非法

    # 应用掩码
    logits = logits.masked_fill(bitmask, float('-inf'))

    token_id = torch.multinomial(probs, 1).item()

    # 更新语法状态
    grammar.accept_token(token_id)

    return token_id
```

### 5.4 增量解码

```python
def decode_incremental(
    self,
    token_ids: List[int],
    status: DecodeStatus,
) -> str:
    """
    增量解码 - 只解码新增的部分。

    用于流式输出，避免重复解码整个序列。
    """

    # 获取新增的 token IDs
    new_token_ids = token_ids[len(status.decode_ids):]

    if not new_token_ids:
        return ""

    # 解码新 tokens
    new_text = self.tokenizer.decode(
        new_token_ids,
        skip_special_tokens=True,
    )

    # 更新状态
    status.decode_ids = token_ids.copy()
    status.decoded_text += new_text

    return new_text
```

```python
async def handle_generate_request(
    self,
    obj: GenerateReqInput,
    request: Request,
):
    """处理生成请求 (支持流式)"""

    if obj.stream:
        # 流式响应
        async for output in self._wait_one_response(obj, state, request):
            yield output.__dict__  # SSE 格式
    else:
        # 非流式响应
        output = await self._wait_final_response(obj, state, request)
        yield output.__dict__


async def _wait_one_response(
    self,
    obj: GenerateReqInput,
    state: ReqState,
    request: Request,
):
    """等待流式响应"""
    state.event = asyncio.Event()

    while True:
        # 等待新输出
        await state.event.wait()

        while state.out_list:
            output = state.out_list.pop(0)
            yield output

            if output.finished:
                return

        state.event.clear()
```

### 7.3 推测解码流程

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                           推测解码流程                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  1. Draft 模型快速生成 K 个候选 tokens                                        │
│     └─ draft_output = draft_model.forward_decode(batch)                     │
│                                                                             │
│  2. Target 模型并行验证                                                       │
│     └─ target_output = target_model.forward_verify(draft_tokens)            │
│                                                                             │
│  3. 比较并接受正确的 tokens                                                    │
│     for i in range(K):                                                      │
│         if draft_tokens[i] == target_tokens[i]:                             │
│             accept token                                                    │
│         else:                                                               │
│             break and resample                                              │
│                                                                             │
│  4. 可能额外接受一个 token (从 Target 的采样)                                  │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
```

### 7.4 性能收益

```
标准解码：每个 token 需要一次大模型前向传播

推测解码 (K=4)：
- 最佳情况：5 个 tokens / 2 次大模型前向传播 (草稿 + 验证)
- 平均情况：2-3 个 tokens / 2 次前向传播

加速比：通常 1.5x - 2.5x
```

```json
        },
        "prompt_tokens": 6,
        "completion_tokens": 12,
        "cached_tokens": 4
    }
}
```

## 九、总结

### 9.1 采样与输出流程

```
Logits [batch_size, vocab_size]
       │
       ├── Logit Bias
       ├── Penalties (frequency/presence/repetition)
       ├── Grammar Mask
       │
       ▼
Filtered Logits
       │
       ├── Temperature
       ├── Top-k Filter
       ├── Top-p Filter
       ├── Min-p Filter
       │
       ▼
Probs [batch_size, vocab_size]
       │
       └── Multinomial Sample
              │
              ▼
       Sampled Token IDs
              │
              ▼
       DetokenizerManager
              │
              ├── Incremental Decode
              ├── Stop String Detection
              │
              ▼
       Output Text (stream/batch)
```

### 9.2 关键设计

1. **灵活的采样参数**：支持多种采样方法和惩罚机制
2. **约束解码**：JSON Schema、正则、EBNF 等语法约束
3. **增量解码**：高效的流式输出实现
4. **推测解码**：EAGLE 等加速技术
