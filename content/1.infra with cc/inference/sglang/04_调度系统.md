## 三、事件循环

### 3.1 正常事件循环

```python
def event_loop_normal(self):
    """正常的调度事件循环"""
    while True:
        # 1. 接收请求
        recv_reqs = self.recv_requests()
        self.process_input_requests(recv_reqs)

        # 2. 获取下一个要运行的批次
        batch = self.get_next_batch_to_run()
        self.cur_batch = batch

        # 3. 执行批次
        if batch:
            result = self.run_batch(batch)
            self.process_batch_result(batch, result)
        else:
            # 空闲时执行检查
            self.self_check_during_idle()

        # 4. 更新状态
        self.last_batch = batch
```

### 3.2 重叠事件循环

重叠模式实现 CPU-GPU 流水线：

```python
def event_loop_overlap(self):
    """CPU-GPU 重叠的调度循环"""
    self.result_queue: Deque[Tuple[ScheduleBatch, BatchResult]] = deque()

    while True:
        # 1. 接收请求
        recv_reqs = self.recv_requests()
        self.process_input_requests(recv_reqs)

        # 2. 获取下一步批次
        batch = self.get_next_batch_to_run()

        # 3. 如果禁用重叠，立即处理上一批次
        if disable_overlap_for_batch:
            self._pop_and_process_result()

        # 4. 执行当前批次
        if batch:
            batch_result = self.run_batch(batch)
            self.result_queue.append((batch.copy(), batch_result))

        # 5. 处理上一批次结果 (延迟一帧)
        if self.last_batch and not disable_overlap_for_batch:
            self._pop_and_process_result()

        # 6. 采样 (依赖上一批次结果)
        self.launch_batch_sample_if_needed(batch_result)

        self.last_batch = batch
```

### 3.3 事件循环选择

SGLang 根据当前配置（并行策略、分离式模式、编译模式等）选择对应的事件循环实现：

```python
def run_event_loop(self):
    """根据配置选择事件循环"""

    if self.disaggregation_mode == DisaggregationMode.NULL:
        if self.enable_pdmux:
            self.event_loop_pdmux()           # PD 多路复用模式
        elif self.pp_size > 1:
            self.event_loop_pp()              # 流水线并行模式
        elif self.enable_overlap:
            self.event_loop_overlap()         # CPU-GPU 重叠模式
        else:
            self.event_loop_normal()          # 普通模式

    elif self.disaggregation_mode == DisaggregationMode.PREFILL:
        self.event_loop_disagg_prefill()      # 分离式 Prefill 节点

    elif self.disaggregation_mode == DisaggregationMode.DECODE:
        self.event_loop_disagg_decode()       # 分离式 Decode 节点

    # 其他特殊模式
    # self.event_loop_dllm_extend()           # Diffusion LLM 扩展
```

**各事件循环特点**：

| 事件循环 | 触发条件 | 核心差异 |
|---------|---------|---------|
| `normal` | 默认 | 单步执行：接收→构建→运行→处理 |
| `overlap` | `enable_overlap=True` | 延迟一帧处理结果，隐藏 CPU 开销 |
| `pp` | `pp_size > 1` | Stage 间 P2P 通信，中间 Stage 只传 hidden states |
| `pdmux` | `enable_pdmux=True` | 混合 Prefill/Decode 的多路调度 |
| `disagg_prefill` | 分离式 Prefill | 执行 Prefill 后发送 KV Cache 到 Decode 节点 |
| `disagg_decode` | 分离式 Decode | 接收 KV Cache，纯 Decode 循环 |
| `dllm_extend` | Diffusion LLM | 处理 diffusion 风格的扩展步进 |

## 五、批处理构建

### 5.1 主入口

```python
def get_next_batch_to_run(self) -> Optional[ScheduleBatch]:
    """获取下一个要运行的批次"""

    # 1. 超时检查
    self._abort_on_waiting_timeout()
    self._abort_on_running_timeout()

    # 2. 处理分块请求
    if self.chunked_req is not None:
        self.stash_chunked_request(self.chunked_req)

    # 3. 将上一轮 prefill 合并到 running batch
    if self.last_batch and self.last_batch.forward_mode.is_extend():
        self.last_batch.filter_batch(...)

        if not self.last_batch.is_empty():
            self.running_batch.merge_batch(self.last_batch)

    # 4. 获取新的 prefill 批次或运行 decode
    new_batch = self.get_new_batch_prefill()

    if new_batch is not None:
        return new_batch
    else:
        # 运行 decode 批次
        if not self.running_batch.is_empty():
            self.running_batch = self.update_running_batch(self.running_batch)
            return self.running_batch

    return None
```

### 5.2 Prefill 批次构建

```python
def _get_new_batch_prefill_raw(self, prefill_delayer):
    """构建 prefill 批次"""

    # 1. 检查语法是否就绪
    if self.grammar_manager.has_waiting_grammars():
        ready_reqs = self.grammar_manager.get_ready_grammar_requests()
        for req in ready_reqs:
            self._add_request_to_queue(req)

    # 2. 检查是否已满或队列为空
    if self.running_batch.batch_is_full or len(self.waiting_queue) == 0:
        return None

    # 3. 计算优先级
    self.policy.calc_priority(self.waiting_queue, self.running_batch)

    # 4. 创建 PrefillAdder (预算管理器)
    adder = PrefillAdder(
        self.page_size,
        self.tree_cache,
        self.token_to_kv_pool_allocator,
        self.running_batch,
        self.new_token_ratio,
        self.max_prefill_tokens,
        self.chunked_prefill_size,
    )

    # 5. 先处理分块请求
    if self.chunked_req is not None:
        self.chunked_req.init_next_round_input()
        self.chunked_req = adder.add_chunked_req(self.chunked_req)

    # 6. 从等待队列添加请求
    for req in self.waiting_queue:
        if self.running_batch.batch_is_full:
            if not self.enable_priority_preemption:
                break

        # 初始化下一轮输入
        req.init_next_round_input(self.tree_cache)

        # 添加到批次
        res = adder.add_one_req(req, has_chunked_req, truncation_align_size)

        if res != AddReqResult.CONTINUE:
            break

    # 7. 创建批次
    can_run_list = adder.can_run_list
    new_batch = ScheduleBatch.init_new(can_run_list, ...)
    new_batch.prepare_for_extend()

    # 8. 混合分块 prefill
    if self.is_mixed_chunk and not self.running_batch.is_empty():
        new_batch.mix_with_running(self.running_batch)

    return new_batch
```

### 5.3 PrefillAdder 预算管理

```python
class PrefillAdder:
    """Prefill 批次预算管理器"""

    def __init__(self, page_size, tree_cache, allocator, running_batch, ...):
        self.page_size = page_size
        self.tree_cache = tree_cache
        self.allocator = allocator
        self.running_batch = running_batch

        self.can_run_list: List[Req] = []
        self.rem_total_token_offset = 0

    @property
    def rem_total_tokens(self) -> int:
        """剩余可用 token 数"""
        available = self.allocator.available_size()
        evictable = self.tree_cache.evictable_size()
        return available + evictable - self.rem_total_token_offset

    def add_one_req(self, req: Req, has_chunked_req: bool, ...):
        """添加一个请求到批次"""

        # 计算需要的 token 数
        prefix_len = len(req.prefix_indices)
        input_len = req.extend_input_len
        total_tokens = self._compute_total_tokens(req)

        # 检查预算
        if total_tokens >= self.rem_total_tokens:
            return AddReqResult.NO_TOKEN

        # 锁定缓存节点 (防止被淘汰)
        with self._lock_node(req.last_node):
            self._update_prefill_budget(prefix_len, input_len, ...)
            self.can_run_list.append(req)

        return self._check_budget_state()
```

### 8.2 处理 Prefill 结果

```python
def process_batch_result_prefill(self, batch: ScheduleBatch, result: BatchResult):
    """处理 prefill 结果"""

    # 1. 等待 GPU->CPU 拷贝完成
    if result.copy_done is not None:
        result.copy_done.synchronize()

    # 2. 获取采样的 token IDs
    next_token_ids = result.sample_output.sampled_token_ids.cpu().tolist()

    # 3. 处理每个请求
    for i, (req, next_token_id) in enumerate(zip(batch.reqs, next_token_ids)):
        if req.finished() or req.is_retracted:
            continue

        if req.is_chunked <= 0:
            # Prefill 完成
            req.time_stats.set_prefill_finished_time()
            req.output_ids.append(next_token_id)
            req.check_finish()

            if req.finished():
                # 释放 KV Cache
                release_kv_cache(req, self.tree_cache)
                req.time_stats.set_completion_time()
            else:
                # 缓存未完成的请求
                self.tree_cache.cache_unfinished_req(req)
        else:
            # 分块 prefill 进行中
            req.is_chunked -= 1

    # 4. 流式输出
    self.stream_output(batch.reqs, batch.return_logprob)
```

```python
def process_batch_result_decode(self, batch: ScheduleBatch, result: BatchResult):

            req.time_stats.set_completion_time()

        # 处理对数概率

            req.output_token_logprobs_val.append(...)

        # 更新语法状态
        if req.grammar is not None:
            req.grammar.accept_token(next_token_id)

    # 3. 流式输出
    self.stream_output(batch.reqs, batch.return_logprob)
```

## 九、调度策略

### 9.1 SchedulePolicy

**文件位置**：`python/sglang/srt/managers/schedule_policy.py`

```python
class SchedulePolicy:
    """调度策略"""

    def __init__(self, policy: str, tree_cache: BasePrefixCache):
        self.policy = self._validate_policy(policy)
        self.tree_cache = tree_cache

        # 批内前缀缓存的模拟 radix tree
        self.waiting_queue_radix_tree = RadixCache.create_simulated()

    def calc_priority(
        self,
        waiting_queue: List[Req],
        running_batch: Optional[ScheduleBatch] = None,
    ):
        """计算并排序优先级"""

        if self.policy == CacheAwarePolicy.LPM:
            # 最长前缀匹配优先
            self._compute_prefix_matches(waiting_queue)
            waiting_queue.sort(key=lambda r: -len(r.prefix_indices))

        elif self.policy == CacheAwarePolicy.DFS_WEIGHT:
            # DFS 权重排序
            self._sort_by_dfs_weight(waiting_queue, self.tree_cache)

        elif self.policy == CacheAgnosticPolicy.FCFS:
            # 先到先服务 (默认顺序)
            pass

        elif self.policy == CacheAgnosticPolicy.LOF:
            # 最长输出优先
            waiting_queue.sort(key=lambda r: -r.sampling_params.max_new_tokens)

        elif self.policy == CacheAgnosticPolicy.RANDOM:
            # 随机
            random.shuffle(waiting_queue)
```

### 9.2 前缀匹配计算

```python
def _compute_prefix_matches(self, waiting_queue: List[Req]):
    """计算前缀匹配 (LPM 策略)"""

    temporary_deprioritized = set()
    self.waiting_queue_radix_tree.reset()

    for r in waiting_queue:
        # 匹配现有缓存
        match_result = self.tree_cache.match_prefix(
            key=r.origin_input_ids + r.output_ids,
            ...
        )
        r.prefix_indices = match_result.device_indices
        r.last_node = match_result.last_device_node

        # 批内前缀缓存：降低共享前缀请求的优先级
        if len(r.prefix_indices) <= THRESHOLD:
            in_batch_match = self.waiting_queue_radix_tree.match_prefix(...)
            if len(in_batch_match) >= DEPRIORITIZE_THRESHOLD:
                temporary_deprioritized.add(r.rid)
            else:
                self.waiting_queue_radix_tree.insert(...)

    return temporary_deprioritized
```

### 9.3 策略对比

| 策略 | 描述 | 优点 | 缺点 |
|------|------|------|------|
| `fcfs` | 先到先服务 | 公平、简单 | 不考虑缓存复用 |
| `lpm` | 最长前缀匹配优先 | 高缓存命中率 | 可能饥饿 |
| `dfs-weight` | DFS 权重 | 复杂场景优化 | 计算开销大 |
| `lof` | 最长输出优先 | 高吞吐 | 短请求延迟大 |

```python
        # 缓存相关
        self.req_pool_idx: Optional[int] = None     # 在 req_to_token_pool 中的索引
        self.prefix_indices: torch.Tensor = ...     # 前缀匹配的 token 索引
        self.last_node: TreeNode = None             # 树中最后一个匹配节点

        # 前向控制
        self.extend_input_len = 0                   # 本轮需要 prefill 的长度
        self.is_chunked = 0                         # 分块计数

        # 完成状态
        self.finished_reason: Optional[BaseFinishReason] = None

        # 统计
        self.time_stats = ReqTimeStats()

    @property
    def seqlen(self) -> int:
        return len(self.origin_input_ids) + len(self.output_ids)

    def init_next_round_input(self, tree_cache):
        """准备下一轮输入"""
        self.fill_ids = self.origin_input_ids + self.output_ids

        if tree_cache is not None:
            match = tree_cache.match_prefix(RadixKey(self.fill_ids))
            self.prefix_indices = match.device_indices
            self.last_node = match.last_device_node

        self.extend_input_len = len(self.fill_ids) - len(self.prefix_indices)
```

### 11.2 Batch Overlap

**文件位置**：`python/sglang/srt/batch_overlap/`

实现不同粒度的 CPU-GPU 重叠策略：

- **`single_batch_overlap.py`**：在单个 batch 的 CPU 预处理（如语法编译、调度决策）与 GPU 计算之间做 overlap
- **`two_batch_overlap.py`**：更深度的双 batch 流水线，当一个 batch 在 GPU 上执行时，CPU 提前准备下一个 batch 的所有输入张量
- **`operations_strategy.py`**：根据当前负载和 batch 特性，动态选择最优 overlap 策略

```
时间轴 ──────────────────────────────────────────────────────►

single_batch_overlap:
  CPU: [调度决策]                 [结果处理][调度决策]
  GPU:          [前向传播]                        [前向传播]

two_batch_overlap:
  CPU: [调度A]       [调度B]       [调度C]
  GPU:          [执行A]       [执行B]       [执行C]
```

### 12.1 核心流程

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                           调度核心流程                                        │
├─────────────────────────────────────────────────────────────────────────────┤
```

### 11.2 关键设计

1. **连续批处理**：动态添加/移除请求，最大化 GPU 利用率
2. **前缀感知调度**：LPM 策略最大化缓存复用
3. **分块 Prefill**：大请求分块处理，减少尾部延迟
4. **CPU-GPU 重叠**：延迟一帧处理结果，隐藏 CPU 开销
5. **多模式支持**：统一框架处理 prefill、decode、推测解码等
