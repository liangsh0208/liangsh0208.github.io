## 1. 模块概述

流水线并行（Pipeline Parallel, PP）在层间切分模型，是跨节点训练的核心技术。

### 【重点】调度策略对比

| 策略 | 实现文件 | 特点 | 适用场景 |
|------|---------|------|---------|
| **1F1B** | `schedules.py:forward_backward_pipelining_without_interleaving` | 标准1前1后，内存高效 | 通用场景 |
| **Interleaved 1F1B** | `schedules.py:forward_backward_pipelining_with_interleaving` | 交错多chunk，气泡最小 | 层数多、显存足 |
| **Virtual PP** | 同上 | 逻辑切分多stage | 层数/PP数不匹配 |
| **Combined 1F1B** | `combined_1f1b.py` | 1F1B变体优化 | 特定CP场景 |

**气泡公式**: `bubble_fraction = (pp_size - 1) / (vpp_size × num_microbatches)`

### 3.1 调度函数选择器

```python

def get_forward_backward_func(
    virtual_pipeline_model_parallel_size: int,
    pipeline_model_parallel_size: int,
    no_pipelining: bool = False,
):
    """根据配置自动选择合适的PP调度策略。
    
    Returns:
        调度函数，接受 (forward_step_func, data_iterator, model, ...) 参数
    """

        # 无PP: 简单的前向后向
        return forward_backward_no_pipelining
    
    if virtual_pipeline_model_parallel_size is not None:
        # 启用Virtual PP: 交错调度
        return forward_backward_pipelining_with_interleaving

    return forward_backward_pipelining_without_interleaving
```

### 3.2 1F1B核心实现

```python
# 文件: megatron/core/pipeline_parallel/schedules.py (简化逻辑)

def forward_backward_pipelining_without_interleaving(
    forward_step_func,
    data_iterator,
    model,
    num_microbatches: int,
    forward_only: bool = False,
):
    """标准1F1B流水线调度。
    
    分为三阶段:
    1. Warmup: 连续前向，填充流水线
    2. Steady: 前向+后向交错 (1F1B)
    3. Cooldown: 连续后向，排空流水线
    """
    pp_rank = get_pipeline_model_parallel_rank()
    pp_size = get_pipeline_model_parallel_world_size()
    
    # 计算各阶段microbatch数
    num_warmup_microbatches = pp_size - 1  # 需要填满后面的stages
    num_steady_microbatches = num_microbatches - num_warmup_microbatches
    
    input_tensors = []  # 保存前向输入用于反向
    output_tensors = []  # 保存前向输出用于反向
    
    # ========== STAGE 1: WARMUP (只前向) ==========
    for i in range(num_warmup_microbatches):
        if not is_first_stage():
            # 非首stage: 从前一stage接收输入
            input_tensor = p2p_communication.recv_forward(tensor_shape)
        else:
            # 首stage: 从data_iterator获取
            input_tensor = forward_step_func.get_batch(data_iterator)
            
        # 执行前向
        output_tensor = forward_step_func(input_tensor, model)
        
        if not is_last_stage():
            # 非尾stage: 发送到下一stage
            p2p_communication.send_forward(output_tensor)
            
        # 保存用于反向
        input_tensors.append(input_tensor)
        output_tensors.append(output_tensor)
    
    # ========== STAGE 2: 1F1B STEADY (交替前向后向) ==========
    for i in range(num_steady_microbatches):
        # 1 Forward
        if is_first_stage():
            input_tensor = forward_step_func.get_batch(data_iterator)
        else:
            input_tensor = p2p_communication.recv_forward(tensor_shape)
            
        output_tensor = forward_step_func(input_tensor, model)
        
        if not is_last_stage():
            p2p_communication.send_forward(output_tensor)
            
        input_tensors.append(input_tensor)
        output_tensors.append(output_tensor)
        
        # 1 Backward (弹出最早保存的微批)
        input_tensor = input_tensors.pop(0)
        output_tensor = output_tensors.pop(0)
        
        if not is_last_stage():
            # 非尾stage: 从后一stage接收梯度
            grad_output = p2p_communication.recv_backward(tensor_shape)
        else:
            grad_output = None
            
        # 执行反向
        grad_input = backward_step_func(
            input_tensor, output_tensor, grad_output
        )
        
        if not is_first_stage():
            # 非首stage: 发送梯度到前一stage
            p2p_communication.send_backward(grad_input)
    
    # ========== STAGE 3: COOLDOWN (只后向) ==========
    for i in range(num_warmup_microbatches):
        input_tensor = input_tensors.pop(0)
        output_tensor = output_tensors.pop(0)
        
        if not is_last_stage():
            grad_output = p2p_communication.recv_backward(tensor_shape)
            
        grad_input = backward_step_func(
            input_tensor, output_tensor, grad_output
        )
        
        if not is_first_stage():
            p2p_communication.send_backward(grad_input)
```

```python
def _communicate(tensor_send_next, tensor_recv_prev, ...):
    """底层P2P通信实现。
    
    使用P2P优化:
    1. 双向同时通信避免死锁
    2. 预分配固定buffer减少分配开销
    3. batch_isend_irecv批量提交
    """
    ops = []
    if tensor_send_next is not None:
        send_next_op = torch.distributed.P2POp(
            torch.distributed.isend,
            tensor_send_next,
            get_pipeline_model_parallel_next_rank(),
        )
        ops.append(send_next_op)
        
    if tensor_recv_prev is not None:
        recv_prev_op = torch.distributed.P2POp(
            torch.distributed.irecv,
            tensor_recv_prev,
            get_pipeline_model_parallel_prev_rank(),
        )
        ops.append(recv_prev_op)
        
    # 批量执行(避免顺序阻塞)
    reqs = torch.distributed.batch_isend_irecv(ops)
    for req in reqs:
        req.wait()  # 等待完成
```

### 3.4 激活值卸载 (Activation Checkpointing)

```python
# 文件: megatron/core/pipeline_parallel/fine_grained_activation_offload.py

class FineGrainedActivationOffload:
    """细粒度激活值卸载到CPU。
    
    解决长序列+大PP导致的激活显存爆炸:
    - 前向激活offload到pinned CPU内存
    - 反向时重新load到GPU
    - 与选择性重计算结合使用
    """
    
    def __init__(self, enabled: bool = False, num_prefetch_layers: int = 2):
        self.enabled = enabled
        self.num_prefetch_layers = num_prefetch_layers
        
    def offload_forward_activation(self, tensor, layer_idx):
        """卸载前向激活到CPU"""
        if not self.enabled:
            return tensor
            
        # 创建pinned CPU buffer
        cpu_buffer = torch.empty(
            tensor.shape, 
            dtype=tensor.dtype,
            pin_memory=True,  # 锁页内存加速H2D
            device='cpu'
        )
        cpu_buffer.copy_(tensor, non_blocking=True)
        
        return cpu_buffer  # 返回CPU引用
    
    def prefetch_backward_activation(self, layer_idx):
        """预取即将计算反向的激活"""
        target_idx = layer_idx - self.num_prefetch_layers
        if target_idx in self.offloaded_tensors:
            tensor = self.offloaded_tensors[target_idx]
            # 异步H2D传输
            gpu_tensor = tensor.cuda(non_blocking=True)
            return gpu_tensor
```
