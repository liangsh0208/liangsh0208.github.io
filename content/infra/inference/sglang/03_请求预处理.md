### 3.2 生成请求处理

```python
async def handle_generate_request(
    self,
    request: TokenizedGenerateReqInput,
):
    """处理生成请求"""

    # 1. 创建请求对象
    req = Req(
        rid=request.rid,
        origin_input_ids=request.origin_input_ids,
        sampling_params=request.sampling_params,
        return_logprob=request.return_logprob,
        stream=request.stream,
        ...

    # 2. 处理多模态输入
    if request.image_inputs:
        await self.process_image_inputs(req, request.image_inputs)

    # 3. 处理 LoRA
    if request.lora_path:
        req.lora_id = await self.lora_registry.load_lora(request.lora_path)

    # 4. 处理会话
    if request.session_id:
        self.session_controller.attach_session(req, request.session_id)

    # 5. 发送到 Scheduler
    self.reqs[req.rid] = req
    self.send_to_scheduler.send_pyobj(request)
```

### 4.2 分词过程

```python
def tokenize(self, text: str) -> List[int]:
    """文本分词"""
    return self.tokenizer.encode(text, add_special_tokens=False)

def detokenize(self, token_ids: List[int]) -> str:
    """Token ID 转文本"""
    return self.tokenizer.decode(token_ids, skip_special_tokens=True)
```

### 4.3 多分词器支持

对于某些模型（如 Qwen），支持多分词器模式：

```python
class MultiTokenizerRouter:
    """多分词器路由"""

    def __init__(self, tokenizers: List[Any]):
        self.tokenizers = tokenizers

    def encode(self, text: str, tokenizer_id: int = 0):
        return self.tokenizers[tokenizer_id].encode(text)

    def decode(self, token_ids: List[int], tokenizer_id: int = 0):
        return self.tokenizers[tokenizer_id].decode(token_ids)
```

### 5.2 图片处理流程

```python
async def process_image_inputs(
    self,
    req: Req,
    image_inputs: List,
):
    """处理图片输入"""

    processed_images = []

    for image_input in image_inputs:
        # 1. 加载图片
        if image_input.startswith("data:"):
            # Base64 编码
            image = decode_base64_image(image_input)
        elif image_input.startswith("http"):
            # URL
            image = await fetch_image(image_input)
        else:
            # 本地文件
            image = load_image(image_input)

        # 2. 预处理
        pixel_values = self.image_processor(image)

        # 3. 存储到请求中
        processed_images.append(pixel_values)

    # 4. 更新请求信息
    req.image_inputs = processed_images
```

### 5.3 图片预处理

```python
class ImageProcessor:
    """图片处理器"""

    def __call__(self, image: Image.Image) -> torch.Tensor:
        # 1. 缩放
        image = self.resize(image, self.size)

        # 2. 裁剪
        image = self.center_crop(image, self.crop_size)

        # 3. 归一化
        pixel_values = self.normalize(image, self.mean, self.std)

        # 4. 转换为张量
        return torch.tensor(pixel_values, dtype=torch.float32)
```

实际的多模态处理器针对不同视觉模型有专用实现（CLIP、SigLIP、InternViT 等），位于 `multimodal/processors/` 和 `multimodal/mm_utils.py`。

部分实现还结合了 ViT CUDA Graph 优化（`vit_cuda_graph_runner.py`），在固定输入 shape 下捕获 ViT 前向图以降低 kernel launch 开销。

### 5.4 视频处理

SGLang 支持视频输入，处理流程与图片类似但增加了时序维度：

```
video_data (base64/URL)
       │
       ▼
┌─────────────────┐
│  Video Decoder  │  解码视频帧 (如 ffmpeg)
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  Frame Sampler  │  均匀/自适应采样关键帧
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  Vision Encoder │  逐帧提取视觉特征
│   (ViT/CLIP)    │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ Temporal Fusion │  帧间特征聚合(如 3D CNN / 纯堆叠)
└────────┬────────┘
         │
         ▼
 video_inputs (tensor)
```

视频请求的 `image_data` 字段可以传入视频文件，TokenizerManager 会根据内容类型自动调用 `VideoProcessor`。

### 5.4 特征传输优化

```python
def wrap_shm_features(features: torch.Tensor, mode: TensorTransportMode):
    """
    使用共享内存优化特征传输。

    避免在 TokenizerManager 和 Scheduler 之间复制大张量。
    """
    if mode == TensorTransportMode.SHARED_MEMORY:
        shared_tensor = SharedTensor.from_tensor(features)
        return shared_tensor.metadata  # 只传输元数据
    else:
        return features
```

```python
class SamplingParams:
    """采样参数"""

    def __init__(
        self,
        # ========== 长度控制 ==========
        max_new_tokens: int = 128,           # 最大生成 token 数
        min_new_tokens: int = 0,             # 最小生成 token 数

        stop_token_ids: Optional[List[int]] = None,  # 停止 token ID 列表

        top_p: float = 1.0,                  # Nucleus 采样阈值

        min_p: float = 0.0,                  # Min-p 采样阈值

        # ========== 惩罚参数 ==========
        frequency_penalty: float = 0.0,      # 频率惩罚 (-2 ~ 2)

        structural_tag: Optional[str] = None, # 结构化标签

        # ========== 其他参数 ==========
        n: int = 1,                          # 返回结果数量
        ignore_eos: bool = False,            # 忽略 EOS token
        skip_special_tokens: bool = True,    # 跳过特殊 token
        logit_bias: Optional[Dict[str, float]] = None,  # Logit 偏置
    ):
```

### 6.2 参数校验

```python
def validate_params(self):
    """参数校验"""

    # 温度范围
    if self.temperature < 0:
        raise ValueError("temperature must be non-negative")

    # Top-p 范围
    if not 0 < self.top_p <= 1:
        raise ValueError("top_p must be in (0, 1]")

    # 惩罚参数范围
    if not -2 <= self.frequency_penalty <= 2:
        raise ValueError("frequency_penalty must be in [-2, 2]")

    if self.repetition_penalty <= 0:
        raise ValueError("repetition_penalty must be positive")
```

## 七、约束解码初始化

### 7.1 GrammarManager

**文件位置**：`python/sglang/srt/constrained/grammar_manager.py`

```python
class GrammarManager:
    """
    语法管理器 - 管理约束解码。

    支持：
    - JSON Schema
    - 正则表达式
    - EBNF 语法
    """

    def __init__(self, server_args: ServerArgs):
        self.grammar_backend = create_grammar_backend(
            server_args.grammar_backend
        )
        self.pending_grammars: Dict[str, Future] = {}

    def compile_grammar(
        self,
        key: str,
        json_schema: Optional[str] = None,
        regex: Optional[str] = None,
        ebnf: Optional[str] = None,
    ):
        """编译语法约束"""
        if json_schema:
            return self.grammar_backend.compile_json_schema(json_schema)
        elif regex:
            return self.grammar_backend.compile_regex(regex)
        elif ebnf:
            return self.grammar_backend.compile_ebnf(ebnf)
```

### 7.2 语法编译流程

```python
async def compile_grammar_async(
    self,
    sampling_params: SamplingParams,
) -> Optional[BaseGrammarObject]:
    """异步编译语法"""

    # 根据类型选择编译方法
    if sampling_params.json_schema:
        # JSON Schema 编译
        grammar = await self.grammar_backend.compile_json_schema_async(
            sampling_params.json_schema
        )
    elif sampling_params.regex:
        # 正则表达式编译
        grammar = await self.grammar_backend.compile_regex_async(
            sampling_params.regex
        )
    elif sampling_params.ebnf:
        # EBNF 编译
        grammar = await self.grammar_backend.compile_ebnf_async(
            sampling_params.ebnf
        )
    else:
        grammar = None

    return grammar
```

## 九、会话管理

### 9.1 Session Controller

```python
class SessionController:
    """会话控制器"""

    def __init__(self):
        self.sessions: Dict[str, Session] = {}

    def create_session(self, session_id: str) -> Session:
        """创建会话"""
        session = Session(session_id=session_id)
        self.sessions[session_id] = session
        return session

    def attach_session(self, req: Req, session_id: str):
        """将请求附加到会话"""
        if session_id not in self.sessions:
            self.create_session(session_id)

        session = self.sessions[session_id]
        req.session = session

        # 复用会话中的历史
        req.history = session.history.copy()
```

```python
class ReqTimeStats:
    """请求时间统计"""

    def __init__(self):
        self.created_time: float = time.time()
        self.tokenized_time: Optional[float] = None
        self.prefill_start_time: Optional[float] = None
        self.prefill_finished_time: Optional[float] = None
        self.completion_time: Optional[float] = None

    @property
    def total_latency(self) -> float:
        return self.completion_time - self.created_time

    @property
    def prefill_latency(self) -> float:
        return self.prefill_finished_time - self.prefill_start_time

    @property
    def decode_latency(self) -> float:
```

```python
class TokenizerManager:

        """追踪请求状态"""
```

### 11.2 批量处理

```python
async def process_batch(self, requests: List[GenerateReqInput]):
    """批量处理请求"""

    # 并发处理所有请求
    tasks = [self.process_request(r) for r in requests]
    results = await asyncio.gather(*tasks)
    return results
```
