### 4.2 Top-K 路由内核

**文件**：`moe_topk_softmax_kernels.cu`

```cpp
// MoE Top-K 路由: 为每个 token 选择 top-k 个专家

template<int TOP_K>
__global__ void moe_topk_softmax_kernel(
    const float* __restrict__ gating_output,  // [num_tokens, num_experts]
    float* __restrict__ expert_weights,       // [num_tokens, TOP_K]
    int* __restrict__ expert_indices,         // [num_tokens, TOP_K]
    int num_tokens,
    int num_experts
) {
    int token_id = blockIdx.x;

    // 1. 计算 softmax
    float max_val = -INFINITY;
    for (int e = 0; e < num_experts; e++) {
        float val = gating_output[token_id * num_experts + e];
        max_val = max(max_val, val);
    }

    float sum = 0.0f;
    for (int e = 0; e < num_experts; e++) {
        float val = gating_output[token_id * num_experts + e];
        sum += expf(val - max_val);
    }

    // 2. 选择 Top-K
    // 使用堆或部分排序
    float weights[TOP_K];
    int indices[TOP_K];
    // ... top-k 选择逻辑

    // 3. 写出结果
    for (int k = 0; k < TOP_K; k++) {
        expert_weights[token_id * TOP_K + k] = weights[k];
        expert_indices[token_id * TOP_K + k] = indices[k];
    }
}
```

```cpp
// custom_all_reduce.cu
// 优化的 AllReduce，绕过 NCCL 的一些开销

__global__ void custom_all_reduce_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    float* __restrict__ peer_buffer,
    int size,
    int rank,
    int world_size
) {
    // 1. 写入 peer buffer
    PeerGroup peer_group = get_peer_group();

    for (int i = threadIdx.x; i < size; i += blockDim.x) {
        peer_buffer[rank * size + i] = input[i];
    }

    // 2. 等待所有 rank 写入
    __syncthreads();

    // 3. 读取并求和
    for (int i = threadIdx.x; i < size; i += blockDim.x) {
        float sum = 0;
        for (int r = 0; r < world_size; r++) {
            sum += peer_buffer[r * size + i];
        }
        output[i] = sum;
    }
}
```

### 7.2 融合 RMSNorm 内核

```cpp
// fused_add_rms_norm_kernel.cu
// 融合: output = RMSNorm(residual + input)

__global__ void fused_add_rms_norm_kernel(
    const float* __restrict__ input,
    const float* __restrict__ residual,
    const float* __restrict__ weight,
    float* __restrict__ output,
    float* __restrict__ new_residual,
    int hidden_size,
    float epsilon
) {
    // 1. 加载并相加
    float sum = 0.0f;
    for (int i = threadIdx.x; i < hidden_size; i += blockDim.x) {
        float val = input[i] + residual[i];
        shared_mem[i] = val;
        sum += val * val;
    }

    // 2. 计算 RMS
    __shared__ float rms;
    if (threadIdx.x == 0) {
        rms = rsqrtf(sum / hidden_size + epsilon);
    }
    __syncthreads();

    // 3. 归一化并应用权重
    for (int i = threadIdx.x; i < hidden_size; i += blockDim.x) {
        float val = shared_mem[i] * rms * weight[i];
        output[i] = val;
        new_residual[i] = val;
    }
}
```

## 十一、JIT 内核

### 11.1 JIT 编译机制

**文件位置**：`python/sglang/jit_kernel/`

SGLang 的部分内核采用 JIT（Just-In-Time）编译方式，在首次调用时通过 Triton 动态生成并编译 CUDA 内核。相比预编译的 AOT（Ahead-of-Time）内核，JIT 的优势在于：

1. **灵活适配不同形状**：无需为每种输入尺寸预编译一个内核，Triton 自动处理 tile 划分
2. **快速实验**：修改内核代码后无需重新编译整个 `sgl-kernel` 库
3. **跨平台**：Triton 后端自动适配目标 GPU 架构

```python
class JITKernel:
    """即时编译内核包装"""

    def __init__(self, source: str, function_name: str):
        self.source = source
        self.function_name = function_name
        self._kernel = None
        self._compiled = False

    def compile(self):
        """使用 Triton 编译内核"""
        import triton
        self._kernel = triton.jit(self.source)
        self._compiled = True

    def __call__(self, *args, **kwargs):
        if not self._compiled:
            self.compile()
        return self._kernel(*args, **kwargs)
```

### 11.2 JIT 内核的实际应用场景

在 SGLang 中，JIT Triton 内核主要用于以下场景：

| 场景 | 位置 | 说明 |
|------|------|------|
| RMSNorm | `layers/` | 对 dynamic shape 的 layer norm 做 JIT 融合 |
| 量化/反量化 | `layers/quantization/` | FP8/INT8 Activation 的 per-token 量化 |
| 位置编码 | `layers/rotary_embedding.py` | RoPE 的变长序列逻辑 |
| Custom Op | `jit_kernel/` | 实验性算子和模型特定的融合算子 |

### 11.3 Triton 内核示例

```python
import triton
import triton.language as tl

@triton.jit
def rms_norm_kernel(
    X,  # input
    W,  # weight
    Y,  # output
    stride,
    N,  # hidden size
    eps: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """RMSNorm Triton 内核"""

    row = tl.program_id(0)

    # 计算平方和
    x = tl.load(X + row * stride + tl.arange(0, BLOCK_SIZE))
    sum_sq = tl.sum(x * x, axis=0)

    # 计算 RMS
    rms = tl.rsqrt(sum_sq / N + eps)

    # 归一化并应用权重
    w = tl.load(W + tl.arange(0, BLOCK_SIZE))
    y = (x * rms) * w

    tl.store(Y + row * stride + tl.arange(0, BLOCK_SIZE), y)
```
