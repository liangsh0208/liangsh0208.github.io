## 模块概述

混合精度训练通过结合 FP32（高精度）和 FP16/BF16（低精度、高吞吐）计算，在保持模型精度的同时大幅提升训练速度和显存效率。Accelerate 自动处理混合精度的所有复杂性，包括自动类型转换和梯度缩放。

### 关键设计决策【重点】

| 设计决策 | 说明 | 技术实现 |
|---------|------|---------|
| **自动上下文管理** | forward 自动在 autocast 上下文中执行 | `autocast()` 自动包装 |
| **梯度缩放** | FP16 使用 GradScaler 避免梯度下溢 | `GradScaler` 动态缩放 |
| **BF16 简化** | BF16 不需要梯度缩放 | 自动检测并跳过 scaler |
| **FP8 前沿支持** | 支持 Hopper GPU 的 FP8 | `torch._scaled_mm` |
| **后向转换** | 梯度自动转回 FP32 | `scaled_loss.backward()` |

### 3. 梯度缩放机制详解

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                          GradScaler 工作机制                                  │
└─────────────────────────────────────────────────────────────────────────────┘

初始化: scale = 65536.0 (可配置)

Forward ──► Loss ──► scale * Loss ──► Backward ──► Grad * scale
                                                         │
                                                         ▼
                                              ┌─────────────────────┐
                                              │ Check for Inf/NaN  │
                                              └──────────┬──────────┘
                                                         │
                                           ┌─────────────┴─────────────┐
                                           │                             │
                                           ▼                             ▼
                                      无溢出 (OK)                     溢出 (Inf/NaN)
                                           │                             │
                                           ▼                             ▼
                              ┌──────────────────────────┐    ┌─────────────────────┐
                              │ grad / scale           │    │ Skip optimizer step │
                              │ 转回 FP32              │    │ scale = scale / 2   │
                              │ optimizer.step()       │    │ 不更新权重           │
                              │                        │    │                     │
                              │ if 连续 N 步无溢出:    │    │ if 连续溢出:         │
                              │     scale *= 2        │    │     报错或继续       │
                              └──────────────────────────┘    └─────────────────────┘

Scale Factor 调整示例:

Step 1: Loss=0.001, scale=65536
        scaled_loss=65.5
        Backward OK! ──► scale=65536 (检查后发现无溢出，scale 不变或增大)

Step 2: Loss=0.0001, scale=65536
        scaled_loss=6.55
        Backward OK!

Step 3: scale=131072 (增大后)
        scaled_loss 可能过大
        Backward 发现 Inf/NaN!
        Skip! scale=65536 (回退)
```

### 表意伪代码：自动混合精度实现

```python
class Accelerator:
    def __init__(self, mixed_precision=None, **kwargs):
        """
        初始化混合精度
        """
        self.state = AcceleratorState(mixed_precision=mixed_precision)

        # 初始化梯度缩放器（仅 FP16 需要）
        if self.state.mixed_precision == "fp16":
            from torch.cuda.amp import GradScaler
            self.scaler = GradScaler()
        elif self.state.mixed_precision == "bf16":
            # BF16 不需要 scaler
            self.scaler = None
        else:
            self.scaler = None

    @contextmanager
    def autocast(self):
        """
        自动类型转换上下文

        在模型前向传播时自动调用
        """
        if self.state.mixed_precision == "fp16":
            with torch.cuda.amp.autocast(dtype=torch.float16):
                yield

        elif self.state.mixed_precision == "bf16":
            device_type = self.device.type
            with torch.autocast(device_type=device_type, dtype=torch.bfloat16):
                yield

        else:
            # 无混合精度
            yield

    def backward(self, loss, **kwargs):
        """
        统一的反向传播接口

        自动处理梯度缩放（FP16）或原生 backward（BF16/FP32）
        """
        if self.distributed_type == DistributedType.DEEPSPEED:
            # DeepSpeed 内部处理
            self.deepspeed_engine.backward(loss, **kwargs)

        elif self.scaler is not None:
            # FP16: 使用 GradScaler 缩放后反向传播
            self.scaler.scale(loss).backward(**kwargs)
        else:
            # BF16 / FP32: 直接反向传播
            loss.backward(**kwargs)

    def step(self, optimizer, model=None):
        """
        执行优化器步骤

        FP16 时需要反缩放梯度并检查溢出
        """
        if self.scaler is not None:
            # FP16: 反缩放 + 检查溢出
            self.scaler.step(optimizer)
            self.scaler.update()  # 更新 scale factor

            if self.scaler.get_scale() < initial_scale / 100:
                warnings.warn("Gradient scale has been reduced significantly!")
        else:
            # BF16 / FP32
            optimizer.step()
```

from torch.cuda.amp import GradScaler

class GradScalerKwargs:
    """
    GradScaler 配置参数
    """

        self,
        init_scale: float = 65536.0,     # 初始缩放因子

        backoff_factor: float = 0.5,     # 回退因子

        self.growth_factor = growth_factor
        self.backoff_factor = backoff_factor
        self.growth_interval = growth_interval
        self.enabled = enabled

)

accelerator = Accelerator(

import torch
import torch.nn as nn
import time
from accelerate import Accelerator

def benchmark_mixed_precision(precision="no", num_steps=100):
    """对比不同精度模式下的训练速度和显存占用"""

    print(f"\n{'='*60}")
    print(f"Precision Mode: {precision}")
    print(f"{'='*60}")

    # 初始化 Accelerator
    accelerator = Accelerator(mixed_precision=precision)
    device = accelerator.device

    # 创建较大的模型
    model = nn.Sequential(
        nn.Linear(4096, 8192),
        nn.ReLU(),
        nn.Linear(8192, 4096),
        nn.ReLU(),
        nn.Linear(4096, 1000),
    ).to(device)

    optimizer = torch.optim.AdamW(model.parameters(), lr=1e-3)

    # 模拟数据
    batch_size = 64
    data = torch.randn(batch_size, 4096).to(device)
    target = torch.randint(0, 1000, (batch_size,)).to(device)

    model, optimizer = accelerator.prepare(model, optimizer)

    # 记录初始显存
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats()
        initial_memory = torch.cuda.memory_allocated() / 1e9

    # 热身
    for _ in range(10):
        with accelerator.autocast():
            outputs = model(data)
            loss = nn.functional.cross_entropy(outputs, target)
        accelerator.backward(loss)
        optimizer.step()
        optimizer.zero_grad()

    # 正式计时
    if device.type == "cuda":
        torch.cuda.synchronize()
    start = time.time()

    for _ in range(num_steps):
        with accelerator.autocast():
            outputs = model(data)
            loss = nn.functional.cross_entropy(outputs, target)
        accelerator.backward(loss)
        optimizer.step()
        optimizer.zero_grad()

    if device.type == "cuda":
        torch.cuda.synchronize()
    elapsed = time.time() - start

    # 结果
    if device.type == "cuda":
        peak_memory = torch.cuda.max_memory_allocated() / 1e9
        print(f"峰值显存: {peak_memory:.2f} GB")

    print(f"总时间: {elapsed:.2f}s")
    print(f"每步时间: {elapsed/num_steps*1000:.2f} ms")
    print(f"吞吐量: {num_steps * batch_size / elapsed:.1f} samples/sec")

    if device.type == "cuda":
        print(f"混合精度加速比: ~2-8x (取决于 GPU 和模型)")

    return elapsed

# 对比测试

"""
梯度缩放调参指南
演示不同 scaler 配置对训练的影响
"""
import torch
import torch.nn as nn
from accelerate import Accelerator
from accelerate.utils import GradScalerKwargs

def train_with_scaler_config(config_name, scaler_kwargs, num_steps=1000):
    """使用不同的 GradScaler 配置训练"""

    print(f"\n{'='*60}")
    print(f"Config: {config_name}")
    print(f"{'='*60}")

    accelerator = Accelerator(
        mixed_precision="fp16",
        kwargs_handlers=[scaler_kwargs]
    )

    # 简单模型
    model = nn.Linear(1000, 10)
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

    model, optimizer = accelerator.prepare(model, optimizer)

    skipped_steps = 0

    for step in range(num_steps):
        # 模拟梯度不稳定的数据
        data = torch.randn(32, 1000) * (1 + step / 100)  # 逐渐增大
        target = torch.randint(0, 10, (32,))

        with accelerator.autocast():
            outputs = model(data)
            loss = nn.functional.cross_entropy(outputs, target)

        accelerator.backward(loss)

        # 检查是否跳过了优化步骤
        if accelerator.optimizer_step_was_skipped:
            skipped_steps += 1

        optimizer.step()
        optimizer.zero_grad()

    # 获取最终的 scale
    final_scale = accelerator.scaler.get_scale()

    print(f"跳过的步数: {skipped_steps}/{num_steps}")
    print(f"最终 scale: {final_scale}")
    print(f"有效步数: {num_steps - skipped_steps}")

    return num_steps - skipped_steps

# 对比不同配置
if __name__ == "__main__":
    # 配置1: 默认配置
    default_scaler = GradScalerKwargs(
        init_scale=65536.0,
        growth_factor=2.0,
        backoff_factor=0.5,
        growth_interval=2000,
    )

    # 配置2: 保守配置（初始 scale 小，回退快）
    conservative_scaler = GradScalerKwargs(
        init_scale=2048.0,      # 较小的初始值
        growth_factor=1.5,      # 缓慢增长
        backoff_factor=0.25,    # 激进回退
        growth_interval=1000,   # 更快增长检查
    )

    # 配置3: 激进配置（初始 scale 大，增长快）
    aggressive_scaler = GradScalerKwargs(
        init_scale=131072.0,    # 较大的初始值
        growth_factor=2.0,
        backoff_factor=0.5,
        growth_interval=100,
    )

    # 运行对比
    default_steps = train_with_scaler_config("Default (65536)", default_scaler)
    conservative_steps = train_with_scaler_config("Conservative (2048)", conservative_scaler)
    aggressive_steps = train_with_scaler_config("Aggressive (131072)", aggressive_scaler)

    print("\n" + "="*60)
    print("总结")
    print("="*60)
    print(f"Default:     {default_steps} 有效步")
    print(f"Conservative: {conservative_steps} 有效步")
    print(f"Aggressive:   {aggressive_steps} 有效步")
    print("\n建议：")
    print("- 如果训练初期出现大量梯度溢出，使用保守配置")
    print("- 如果训练稳定但速度较慢，使用激进配置")
    print("- 通常默认配置适用于大多数场景")
