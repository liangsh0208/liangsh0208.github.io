# 06_模型实现

## 1. 目标论证

**为什么要理解模型实现？**

picotron实现了标准Llama架构，但融入了特定的分布式优化：
- Attention支持GQA(Grouped Query Attention)减少KV缓存
- RMSNorm的多种实现(标准、Triton加速)
- 模块设计考虑4D并行切分需求

理解模型层结构有助于调试分布式问题。

## 2. 工作假设

**架构约束：**
- 仅支持Llama/Dense Transformer结构
- 需要GQA配置(MLKV，多查询注意力变体)
- Flash Attention或PyTorch SDPA后端

**分布式假设：**
- 参数需支持TP切分标记
- 某些层可能已被Distributed层包装

### 3.2 Attention层实现

```python
class Attention(nn.Module):
    def __init__(self, config, layer_idx):
        super().__init__()
        self.layer_idx = layer_idx
        self.config = config
        
        # TP切分后的本地头数
        pgm = get_process_group_manager()
        tp_size = pgm.tp_world_size if pgm else 1
        
        self.hidden_size = config.hidden_size
        self.num_heads = config.num_attention_heads
        self.head_dim = self.hidden_size // self.num_heads
        self.num_key_value_heads = config.num_key_value_heads
        
        # TP切分
        self.num_local_heads = self.num_heads // tp_size
        self.num_local_kv_heads = self.num_key_value_heads // tp_size
        self.q_proj = nn.Linear(...)
        self.k_proj = nn.Linear(...)
        self.v_proj = nn.Linear(...)
        self.o_proj = nn.Linear(...)
        
        # RoPE旋转位置编码
        self.rotary_emb = RotaryEmbedding(config=head_dim)
    
    def forward(self, hidden_states, cos, sin, ...):
        batch_size, seq_len, _ = hidden_states.shape
        
        # 投影并分头
        query_states = self.q_proj(hidden_states).view(
            batch_size, seq_len, self.num_local_heads, self.head_dim
        ).transpose(1, 2)
        
        key_states = self.k_proj(hidden_states).view(
            batch_size, seq_len, self.num_local_kv_heads, self.head_dim
        ).transpose(1, 2)
        
        value_states = self.v_proj(hidden_states).view(
            batch_size, seq_len, self.num_local_kv_heads, self.head_dim
        ).transpose(1, 2)
        
        # 应用RoPE
        query_states, key_states = apply_rotary_pos_emb(
            query_states, key_states, cos, sin
        )
        
        # 三种注意力计算模式
        if use_context_parallel and pgm.cp_world_size > 1:
            # 1. Ring Attention (CP>1)
            attn_output = RingAttentionFunc.apply(
                query_states, key_states, value_states, 
                sm_scale=1.0 / math.sqrt(self.head_dim),
                is_causal=True
            )
        elif use_flash_attention:
            # 2. Flash Attention (推荐)
            attn_output = flash_attn_func(
                query_states, key_states, value_states,
                causal=True
            )
        else:
            # 3. PyTorch SDPA (默认)
            attn_output = F.scaled_dot_product_attention(
                query_states, key_states, value_states,
                is_causal=True
            )
        
        # 合并头并投影
        attn_output = attn_output.transpose(1, 2).contiguous()
        attn_output = attn_output.reshape(batch_size, seq_len, -1)
        attn_output = self.o_proj(attn_output)
        
        return attn_output
```

### 3.3 DecoderLayer

```python
class DecoderLayer(nn.Module):
    """单个Transformer层: Pre-LN + Attention + MLP"""
    
    def __init__(self, config, layer_idx):
        super().__init__()
        self.hidden_size = config.hidden_size
        
        # 自注意力
        self.self_attn = Attention(config, layer_idx)
        self.input_layernorm = RMSNorm(config.hidden_size, eps=config.rms_norm_eps)
        
        # MLP (SwiGLU)
        self.mlp = MLP(config)
        self.post_attention_layernorm = RMSNorm(
            config.hidden_size, eps=config.rms_norm_eps
        )
    
    def forward(self, hidden_states, cos, sin, ...):
        residual = hidden_states
        
        # Pre-LN: 先归一化再计算
        hidden_states = self.input_layernorm(hidden_states)
        hidden_states = self.self_attn(hidden_states, cos, sin, ...)
        
        # 残差连接
        hidden_states = residual + hidden_states
        residual = hidden_states
        
        # MLP子层
        hidden_states = self.post_attention_layernorm(hidden_states)
        hidden_states = self.mlp(hidden_states)
        hidden_states = residual + hidden_states
        
        return hidden_states
```

### 3.4 MLP实现(SwiGLU)

```python
class MLP(nn.Module):
    """
    SwiGLU激活: 
    output = down_proj(silu(gate_proj(x)) * up_proj(x))
    """
    def __init__(self, config):
        super().__init__()
        
        # TP切分后的维度
        pgm = get_process_group_manager()
        tp_size = pgm.tp_world_size if pgm else 1
        
        intermediate_size = config.intermediate_size // tp_size
        
        self.gate_proj = nn.Linear(config.hidden_size, intermediate_size)
        self.up_proj = nn.Linear(config.hidden_size, intermediate_size)
        self.down_proj = nn.Linear(intermediate_size, config.hidden_size)
        self.act_fn = F.silu
    
    def forward(self, x):
        return self.down_proj(
            self.act_fn(self.gate_proj(x)) * self.up_proj(x)
        )
```

### 3.5 RMSNorm实现

```python
# 标准实现
class LlamaRMSNorm(nn.Module):
    def __init__(self, hidden_size, eps=1e-6):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(hidden_size))
        self.variance_epsilon = eps
    
    def forward(self, hidden_states):
        input_dtype = hidden_states.dtype
        hidden_states = hidden_states.to(torch.float32)
        
        # E[x^2]
        variance = hidden_states.pow(2).mean(-1, keepdim=True)
        
        # x / sqrt(var + eps) * weight
        hidden_states = hidden_states * torch.rsqrt(
            variance + self.variance_epsilon
        )
        
        return self.weight * hidden_states.to(input_dtype)

# Triton加速版(FlashAttn=1时)
class TritonRMSNorm(nn.Module):
    """融合RMSNorm + 残差连接，减少kernel launch次数"""
    def forward(self, hidden_states, residual=None):
        return layer_norm_fn(
            hidden_states,
            self.weight,
            residual=residual,
            eps=self.variance_epsilon,
            is_rms_norm=True
        )
```

### 3.6 完整Llama模型

```python
class LlamaModel(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.config = config
        
        self.embed_tokens = nn.Embedding(
            config.vocab_size, 
            config.hidden_size
        )
        
        # 创建所有层(后续PP会切分各rank保留的层)
        self.layers = nn.ModuleList([
            DecoderLayer(config, i) 
            for i in range(config.num_hidden_layers)
        ])
        
        self.norm = RMSNorm(config.hidden_size, eps=config.rms_norm_eps)
    
    def forward(self, input_ids):
        hidden_states = self.embed_tokens(input_ids)
        
        # 预计算RoPE cos/sin
        cos, sin = self.rotary_emb(hidden_states, seq_len=seq_len)
        
        for layer in self.layers:
            hidden_states = layer(hidden_states, cos, sin)
        
        hidden_states = self.norm(hidden_states)
        return hidden_states

class LlamaForCausalLM(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.model = LlamaModel(config)
        
        # 强制创建独立lm_head
        # 避免与embed_tokens的tied weights问题
        self.lm_head = nn.Linear(
            config.hidden_size, 
            config.vocab_size, 
            bias=False
        )
    
    def forward(self, input_ids, labels=None):
        hidden_states = self.model(input_ids)
        logits = self.lm_head(hidden_states)
        
        loss = None
        if labels is not None:
            # Shift for next-token prediction
            shift_logits = logits[..., :-1, :].contiguous()
            shift_labels = labels[..., 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)),
                shift_labels.view(-1)
            )
        
        return {'loss': loss, 'logits': logits}
```

## 4. 故障模式

| 问题 | 可能原因 | 解决方案 |
|-----|---------|---------|
| 输出尺寸不匹配 | TP切分未正确处理 | 检查TP后输出的维度应为hidden_size |
| 梯度NaN | RMSNorm epsilon过小 | 增大eps到1e-6 |
| Attention输出形状错误 | GQA reshape问题 | 验证num_local_kv_heads计算 |
| 位置编码错误 | RoPE cos/sin缓存尺寸不对 | 确保max_seq_length覆盖实际序列 |
| lm_head与embedding权重共享 | 未强制创建独立lm_head | 检查apply_tensor_parallel中是否分离 |
