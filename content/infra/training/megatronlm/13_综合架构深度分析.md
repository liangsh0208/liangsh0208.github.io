### 3.2 并行策略详解

| 策略 | 变量 | 描述 | 适用场景 |
|------|------|------|----------|
| **Tensor Parallel (TP)** | `tensor_model_parallel_size` | 层内模型并行，切分权重矩阵 | 大模型、显存瓶颈 |
| **Pipeline Parallel (PP)** | `pipeline_model_parallel_size` | 层间模型并行，流水线执行 | 超深网络 |
| **Virtual Pipeline (VP)** | `virtual_pipeline_model_parallel_size` | 交错流水线，减少气泡 | PP 场景优化 |
| **Data Parallel (DP)** | `data_parallel_size` | 数据并行 | 小批量高性能 |
| **Expert Parallel (EP)** | `expert_model_parallel_size` | 专家并行 | MoE 模型 |
| **Context Parallel (CP)** | `context_parallel_size` | 序列并行 | 长序列训练 |
| **Sequence Parallel (SP)** | `sequence_parallel` | 序列维度并行 | TP 内部优化 |

```python
def get_num_layers_to_build(config, vp_stage, pp_rank):
    """计算当前流水线阶段应构建的层数"""
    
    # 自定义 PP 布局
    if config.pipeline_model_parallel_layout is not None:
        return config.pipeline_model_parallel_layout.get_num_layers_to_build(...)
    
    # 不均匀 PP 划分支持
    if config.num_layers_in_first_pipeline_stage is not None:
        if is_first_pp_stage:
            return config.num_layers_in_first_pipeline_stage
    
    # 均匀划分
    num_layers_per_pipeline_rank = num_layers // config.pipeline_model_parallel_size
    
    # VP 交错划分
    if vp_size is not None:
        return num_layers_per_pipeline_rank // vp_size
    
    return num_layers_per_pipeline_rank
```

## 4. 模型架构实现

### 4.1 GPTModel 核心实现

**文件位置**: `megatron/core/models/gpt/gpt_model.py`

```python
class GPTModel(LanguageModule):
    """GPT Transformer 语言模型"""
    
    def __init__(
        self,
        config: TransformerConfig,
        transformer_layer_spec: ModuleSpec,  # 层规格定义
        vocab_size: int,
        max_sequence_length: int,
        pre_process: bool = True,   # 是否包含 embedding 层 (PP 首阶段)
        post_process: bool = True,  # 是否包含 output 层 (PP 末阶段)
        position_embedding_type: Literal['learned_absolute', 'rope', 'mrope', 'yarn', 'none'] = 'learned_absolute',
        mtp_block_spec: Optional[ModuleSpec] = None,  # MTP 块规格
        ...
    ):
        # Embedding 层 (仅在 pre_process 阶段)
        if self.pre_process or self.mtp_process:
            self.embedding = LanguageModelEmbedding(...)
        
        # RoPE 位置编码
        if self.position_embedding_type == 'rope':
            self.rotary_pos_emb = RotaryEmbedding(...)
        elif self.position_embedding_type == 'yarn':
            self.rotary_pos_emb = YarnRotaryEmbedding(...)
        
        # Transformer 解码器
        self.decoder = TransformerBlock(
            config=self.config,
            spec=transformer_layer_spec,
            pre_process=self.pre_process,
            post_process=self.post_process,
        )
        
        # Multi-Token Prediction 块
        if self.mtp_process:
            self.mtp = MultiTokenPredictionBlock(...)
        
        # 输出层 (仅在 post_process 阶段)
        if self.post_process:
            self.output_layer = tensor_parallel.ColumnParallelLinear(...)
```

### 4.2 前向传播流程

```python
def forward(self, input_ids, position_ids, attention_mask, ...):
    # 1. 预处理: Embedding + 位置编码
    preproc_output = self._preprocess(input_ids, position_ids, ...)
    decoder_input, rotary_pos_emb, ... = preproc_output
    
    # 2. Transformer 解码器
    hidden_states = self.decoder(
        hidden_states=decoder_input,
        attention_mask=attention_mask,
        rotary_pos_emb=rotary_pos_emb,
        ...
    )
    
    # 3. 后处理: MTP + 输出层 + 损失计算
    return self._postprocess(hidden_states, labels, ...)
```

### 5.2 MoE 层实现

```python
class MoELayer(BaseMoELayer):
    """Mixture of Experts 层"""
    
    def __init__(self, config, layer_number, ...):
        # 路由器
        self.router = TopKRouter(config=config, ...)
        
        # Token 分发器
        if token_dispatcher_type == "allgather":
            self.token_dispatcher = MoEAllGatherTokenDispatcher(...)
        else:
            self.token_dispatcher = MoEAlltoAllTokenDispatcher(...)
        
        # 专家网络
        self.experts = TEGroupedMLP(
            num_local_experts=self.num_local_experts,
            config=config,
            ...
        )
        
        # 共享专家 (可选)
        if self.use_shared_expert:
            self.shared_experts = SharedExperts(...)
    
    def forward(self, hidden_states):
        # 1. 路由计算
        scores, routing_map = self.router(hidden_states)
        
        # 2. Token 分发
        dispatched_input, tokens_per_expert = self.token_dispatcher.token_permutation(
            hidden_states, scores, routing_map
        )
        
        # 3. 专家计算
        expert_output, _ = self.experts(dispatched_input, tokens_per_expert, ...)
        
        # 4. Token 归约
        output = self.token_dispatcher.token_unpermutation(expert_output, ...)
        
        # 5. 共享专家 (可选)
        if self.use_shared_expert:
            shared_output = self.shared_experts(hidden_states)
            output = output + shared_output
        
        return output
```

### 5.3 TopKRouter 路由器

```python
class TopKRouter(Router):
    """Top-K 路由器 - 每个 token 选择 top-k 个专家"""
    
    def __init__(self, config, ...):
        self.topk = config.moe_router_topk
        self.routing_type = config.moe_router_load_balancing_type  # aux_loss/none/sinkhorn
        self.score_function = config.moe_router_score_function  # softmax/sigmoid
        
        # 路由权重
        self.weight = torch.nn.Parameter(
            torch.empty((config.num_moe_experts, config.hidden_size))
        )
    
    def forward(self, hidden_states):
        # 1. 计算 logits
        logits = self.gating(hidden_states)  # [batch*seq, num_experts]
        
        # 2. Top-K 选择
        topk_scores, topk_indices = torch.topk(logits, self.topk, dim=-1)
        
        # 3. Score 函数 (softmax/sigmoid)
        scores = self.score_function(topk_scores)
        
        # 4. 辅助损失计算 (负载均衡)
        if self.routing_type == "aux_loss":
            aux_loss = self.compute_aux_loss(logits, topk_indices)
            save_to_aux_losses_tracker(aux_loss, ...)
        
        return scores, topk_indices
```

```python
class ColumnParallelLinear(torch.nn.Module):
    """列并行线性层 - 按输出维度切分"""
    
    def __init__(self, input_size, output_size, config, ...):
        # 每个 TP rank 持有 output_size // world_size 的权重
        self.weight = Parameter(torch.empty(
            self.output_size_per_partition,
            input_size,
            ...
        ))
    
    def forward(self, input):
        # 输入: [S, B, H]
        # 输出: [S, B, H/TP] (每个 rank)
        output = torch.matmul(input, self.weight.t())
        
        if self.gather_output:
            # All-gather 收集所有 TP rank 的输出
            output = gather_from_tensor_parallel_region(output)
        
        return output


class RowParallelLinear(torch.nn.Module):
    """行并行线性层 - 按输入维度切分"""
    
    def forward(self, input):
        # 输入: [S, B, H/TP] (每个 rank)
        output = torch.matmul(input, self.weight.t())
        
        # All-reduce 聚合所有 TP rank 的结果
        output = reduce_from_tensor_parallel_region(output)
        
        return output
```

### 8.2 MegatronOptimizer 基类

```python
class MegatronOptimizer(ABC):
    """Megatron 优化器基类"""
    
    def __init__(self, optimizer, config, init_state_fn):
        self.optimizer = optimizer
        self.config = config
        self.init_state_fn = init_state_fn
    
    def get_main_grads_for_grad_norm(self):
        """获取用于梯度范数计算的梯度 (过滤重复参数)"""
        grads = []
        for param in self.get_parameters():
            if param.grad is not None:
                # 排除共享参数和 TP 副本
                if param_is_not_shared(param) and param_is_not_tensor_parallel_duplicate(param):
                    grads.append(param.grad)
        return grads
    
    def clip_grad_norm(self, max_norm):
        """裁剪梯度范数"""
        grads = self.get_main_grads_for_grad_norm()
        total_norm = get_grad_norm_fp32(grads)
        clip_coef = max_norm / (total_norm + 1e-6)
        if clip_coef < 1.0:
            self._clip_grad_by_norm(clip_coef)
        return total_norm
```

- **梯度分片**: 每个 DP rank 只保存部分参数的梯度
- **状态分片**: 优化器状态 (如 Adam 的 m, v) 分布式存储

```python
def forward_step(data_iterator, model, return_schedule_plan=False):
    """单步前向传播"""
    args = get_args()
    timers = get_timers()
    
    # 1. 获取 batch
    timers('batch-generator').start()
    tokens, labels, loss_mask, attention_mask, position_ids, packed_seq_params = \
        get_batch(data_iterator, vp_stage)
    timers('batch-generator').stop()
    
    # 2. 模型前向
    output_tensor = model(
        tokens, position_ids, attention_mask,
        labels=labels,
        loss_mask=loss_mask,
        packed_seq_params=packed_seq_params,
    )

    return output_tensor, partial(loss_func, loss_mask, model=model)
```

### 9.3 loss_func 函数

```python
def loss_func(loss_mask, output_tensor, model=None):
    """计算损失"""
    losses = output_tensor.view(-1).float()
    loss_mask = loss_mask.view(-1).float()
    
    # 加权求和
    loss = torch.sum(losses * loss_mask)
    num_tokens = loss_mask.sum().clone().detach().to(torch.int)
    
    # 报告指标
    report = {'lm loss': torch.cat([loss.clone().detach().view(1), num_tokens.view(1)])}
    
    # NaN/Inf 检测
    if args.check_for_nan_in_loss_and_grad:
        rerun_state_machine.validate_result(
            result=loss,
            rejection_func=torch.isnan,
            message="found NaN in local forward loss calculation",
        )
    
    return loss, num_tokens, report
```

```python
        self,
```

### 13.1 ModuleSpec 定义

```python
@dataclass
class ModuleSpec:
    """模块规格定义 - 支持动态导入和参数传递"""
    
    module: Union[Tuple, type]  # (模块路径, 类名) 或已导入的类
    params: dict = field(default_factory=lambda: {})  # 初始化参数
    submodules: object = None  # 子模块规格
    metainfo: dict = field(default_factory=lambda: {})  # 元信息
    
    def __call__(self, *args, **kwargs):
        """直接调用以构建实例"""
        return build_module(self, *args, **kwargs)
```

### 13.2 模块构建机制

```python
def build_module(spec_or_module: Union[ModuleSpec, type], *args, **kwargs):
    """从规格或类型构建模块实例"""
    
    # 如果是函数，直接返回
    if isinstance(spec_or_module, types.FunctionType):
        return spec_or_module
    
    # 获取模块类
    if isinstance(spec_or_module, type):
        module = spec_or_module
    elif hasattr(spec_or_module, "module") and isinstance(spec_or_module.module, type):
        module = spec_or_module.module
    else:
        # 动态导入: ("megatron.core.transformer.attention", "SelfAttention")
        module = import_module(spec_or_module.module)
    
    # 传递子模块规格
    if hasattr(spec_or_module, "submodules") and spec_or_module.submodules is not None:
        kwargs["submodules"] = spec_or_module.submodules
    
    # 合并参数并实例化
    return module(
        *args,
        **spec_or_module.params if hasattr(spec_or_module, "params") else {},
        **kwargs
    )
```

### 14.2 TransformerLayer 前向传播

```python
class TransformerLayer(GraphableMegatronModule, BaseTransformerLayer):
    """单个 Transformer 层实现"""
    
    def __init__(
        self,
        config: TransformerConfig,
        submodules: TransformerLayerSubmodules,
        layer_number: int = 1,
        hidden_dropout: Optional[float] = None,
        pg_collection: Optional[ProcessGroupCollection] = None,
        vp_stage: Optional[int] = None,
        is_mtp_layer: bool = False,
    ):
        # 计算全局层编号 (考虑 PP 偏移)
        self.layer_number = layer_number + get_transformer_layer_offset(
            config, vp_stage, get_pg_rank(pg_collection.pp)
        )
        
        # [模块 1]: 输入 LayerNorm
        self.input_layernorm = submodules.input_layernorm(
            config=config, hidden_size=config.hidden_size
        )
        
        # [模块 2]: Self-Attention
        self.self_attention = build_module(
            submodules.self_attention, config=config, layer_number=self.layer_number
        )
        
        # [模块 3]: Bias-Dropout-Add
        self.self_attn_bda = build_module(submodules.self_attn_bda)
        
        # [模块 4-6]: Cross-Attention (可选)
        # ...
        
        # [模块 7]: Pre-MLP LayerNorm
        self.pre_mlp_layernorm = submodules.pre_mlp_layernorm(...)
        
        # [模块 8]: MLP 或 MoE
        self.mlp = build_module(submodules.mlp, config=config, ...)
    
    def forward(
        self,
        hidden_states,
        attention_mask,
        rotary_pos_emb=None,
        inference_context=None,
        packed_seq_params=None,
        **kwargs,
    ):
        # ===== Self-Attention Block =====
        # 1. 输入 LayerNorm
        hidden_states = self.input_layernorm(hidden_states)
        
        # 2. Self-Attention
        hidden_states = self.self_attention(
            hidden_states, attention_mask,
            rotary_pos_emb=rotary_pos_emb,
            inference_context=inference_context,
            packed_seq_params=packed_seq_params,
        )
        
        # 3. Bias-Dropout-Add (残差连接)
        hidden_states = self.self_attn_bda(hidden_states)
        
        # ===== MLP Block =====
        # 4. Pre-MLP LayerNorm
        hidden_states = self.pre_mlp_layernorm(hidden_states)
        
        # 5. MLP 或 MoE
        hidden_states = self.mlp(hidden_states)
        
        # 6. Bias-Dropout-Add (残差连接)
        hidden_states = self.mlp_bda(hidden_states)
        
        return hidden_states
```

### 15.1 DistributedDataParallel 核心实现

```python
class DistributedDataParallel(_BaseDataParallel):
    """
    DDP 包装器，将梯度存储在连续缓冲区中。
    支持梯度分桶和通信-计算重叠。
    """
    
    def __init__(
        self,
        config: TransformerConfig,
        ddp_config: DistributedDataParallelConfig,
        module: torch.nn.Module,
        disable_bucketing: bool = False,
        pg_collection: Optional[ProcessGroupCollection] = None,
    ):
        # 设置进程组
        process_group_dict = ProcessGroupCollection.setup_process_groups_for_ddp(
            pg_collection, config, ddp_config
        )
        
        self.dp_group = process_group_dict['dp_group']
        self.dp_cp_group = process_group_dict['dp_cp_group']
        
        # 桶大小设置
        if ddp_config.bucket_size is None:
            ddp_config.bucket_size = max(40000000, 1000000 * dp_group.size())
        
        # 分组参数
        dense_params = []
        expert_parallel_params = []
        for name, param in self.module.named_parameters():
            if not param.requires_grad:
                continue
            if getattr(param, 'allreduce', True):
                dense_params.append((param, name))
            else:
                expert_parallel_params.append((param, name))
        
        # 分配梯度缓冲区
        self.buffers = self._allocate_buffers_for_parameters(
            dense_params, self.dp_group, gradient_scaling_factor
        )
    
    def forward(self, *inputs, **kwargs):
        """前向传播"""
        return self.module(*inputs, **kwargs)
    
    def start_grad_sync(self):
        """启动梯度同步 (异步)"""
        for buffer in self.buffers:
            buffer.start_grad_sync()
```

## 17. 强化学习 (RL)

**文件位置**: `train_rl.py` 和 `megatron/rl/`

### 17.1 GRPO 训练入口

```python
# train_rl.py
def loss_func(
    loss_mask: torch.Tensor,
    kl_term: torch.Tensor,
    ratios: torch.Tensor,
    entropy_term: torch.Tensor,
    truncated_from_above: torch.Tensor,
    truncated_from_below: torch.Tensor,
    output_tensor: torch.Tensor,
):
    """GRPO 损失函数"""
    
    losses = output_tensor.float()
    loss_mask = loss_mask.float()
    
    # 计算损失
    loss = torch.sum(losses * loss_mask) / loss_mask.sum()
    
    # KL 散度、熵、截断比例等指标
    masked_kl = torch.sum(loss_mask * kl_term)
    masked_entropy = torch.sum(loss_mask * entropy_term)
    
    return loss, total_tokens, {
        'grpo loss': loss,
        'kl': masked_kl,
        'entropy': masked_entropy,
    }
```
