# 实战配置指南

> 本指南提供不同训练场景的配置模板，帮助你快速开始 TRL 训练。

---

## 1. SFT 配置模板

### 1.1 基础 SFT (单卡)

```python
from trl import SFTTrainer, SFTConfig

config = SFTConfig(
    # 输出
    output_dir="./output/sft-base",
    
    # 训练设置
    num_train_epochs=3,
    per_device_train_batch_size=4,
    gradient_accumulation_steps=4,
    learning_rate=2e-5,
    warmup_ratio=0.1,
    lr_scheduler_type="cosine",
    
    # 效率优化
    bf16=True,
    gradient_checkpointing=True,
    dataloader_num_workers=4,
    
    # 日志与保存
    logging_steps=10,
    save_steps=500,
    eval_steps=500,
    eval_strategy="steps",
    load_best_model_at_end=True,
    
    # SFT 特有
    max_length=2048,
    packing=True,
    packing_strategy="bfd",
    completion_only_loss=True,
)

trainer = SFTTrainer(
    model="Qwen/Qwen2.5-0.5B",
    args=config,
    train_dataset=train_dataset,
    eval_dataset=eval_dataset,
)
```

### 1.2 QLoRA SFT (低显存)

```python
from trl import SFTTrainer, SFTConfig, ModelConfig
from peft import get_peft_config

model_config = ModelConfig(
    model_name_or_path="meta-llama/Llama-2-7b",
    use_peft=True,
    lora_r=64,
    lora_alpha=16,
    lora_dropout=0.1,
    lora_target_modules=["q_proj", "v_proj", "k_proj", "o_proj"],
    lora_bias="none",
    lora_task_type="CAUSAL_LM",
    load_in_4bit=True,
    bnb_4bit_compute_dtype="bfloat16",
    bnb_4bit_use_double_quant=True,
    bnb_4bit_quant_type="nf4",
)

config = SFTConfig(
    output_dir="./output/sft-qlora",
    num_train_epochs=3,
    per_device_train_batch_size=1,
    gradient_accumulation_steps=8,
    learning_rate=1e-4,  # LoRA 使用更高学习率
    max_length=512,  # 减轻显存压力
    gradient_checkpointing=True,
)

trainer = SFTTrainer(
    model=model_config.model_name_or_path,
    args=config,
    train_dataset=dataset,
    peft_config=get_peft_config(model_config),
)
```

## 2. DPO 配置模板

### 2.1 基础 DPO

```python
from trl import DPOTrainer, DPOConfig

config = DPOConfig(
    output_dir="./output/dpo-base",
    
    # 训练设置
    num_train_epochs=1,
    per_device_train_batch_size=2,  # DPO 同时处理 chosen+rejected
    gradient_accumulation_steps=8,
    learning_rate=1e-6,  # DPO 使用更低学习率
    warmup_ratio=0.1,
    
    # DPO 特有
    beta=0.1,
    loss_type=["sigmoid"],
    
    # 参考模型优化
    precompute_ref_log_probs=False,
    
    # 效率
    bf16=True,
    gradient_checkpointing=True,
)

# 从 SFT 模型开始
model_name = "your-sft-model/checkpoint-1000"

trainer = DPOTrainer(
    model=model_name,
    args=config,
    train_dataset=preference_dataset,
)
```

### 2.2 MPO (Multi-Loss)

```python
from trl import DPOConfig

config = DPOConfig(
    output_dir="./output/mpo",
    
    # 多 Loss 组合: 80% DPO + 20% SFT
    loss_type=["sigmoid", "sft"],
    loss_weights=[0.8, 0.2],
    
    beta=0.1,
    learning_rate=5e-7,
    per_device_train_batch_size=1,
    gradient_accumulation_steps=8,
)
```

### 2.3 TR-DPO (参考模型同步)

```python
from trl import DPOConfig
from trl.trainer.callbacks import SyncRefModelCallback

config = DPOConfig(
    output_dir="./output/tr-dpo",
    beta=0.1,
    
    # 启用参考模型同步
    sync_ref_model=True,
    ref_model_mixup_alpha=0.5,
    ref_model_sync_steps=64,
    
    # 其他参数
    learning_rate=1e-6,
)

trainer = DPOTrainer(...)
trainer.add_callback(SyncRefModelCallback())
```

## 3. GRPO 配置模板

### 3.1 DeepSeek-R1 风格

```python
from trl import GRPOTrainer, GRPOConfig
from trl.rewards import accuracy_reward, think_format_reward

config = GRPOConfig(
    output_dir="./output/r1-style",
    
    # 生成参数
    num_generations=8,           # G: 每组 8 个 completions
    max_completion_length=2048,   # 长思维链需要更大的空间
    temperature=0.9,
    top_p=0.95,
    
    # 训练参数
    num_train_epochs=1,
    per_device_train_batch_size=4,
    gradient_accumulation_steps=4,
    learning_rate=1e-6,
    warmup_ratio=0.1,
    
    # GRPO 特有
    num_iterations=2,            # μ: 每批次更新 2 次
    epsilon=0.2,                # 裁剪下界
    epsilon_high=0.28,          # 裁剪上界 (DAPO 推荐)
    beta=0.001,                 # 很小的 KL 约束
    
    # Loss 类型
    loss_type="dapo",           # DAPO 算法
    scale_rewards="group",      # 组内标准化
    mask_truncated_completions=True,
    
    # vLLM 加速 (推荐)
    use_vllm=True,
    vllm_mode="colocate",
    vllm_gpu_memory_utilization=0.3,
    vllm_tensor_parallel_size=1,
    
    # 其他
    bf16=True,
    logging_steps=10,
)

trainer = GRPOTrainer(
    model="Qwen/Qwen2.5-0.5B-Instruct",
    reward_funcs=[accuracy_reward, think_format_reward],
    args=config,
    train_dataset=math_dataset,
)
```

## 4. 通用优化配置

### 4.1 显存优化清单

```python
from trl import SFTConfig  # 或 DPOConfig, GRPOConfig

config = SFTConfig(
    # 基本优化 (默认已启用)
    bf16=True,
    gradient_checkpointing=True,
    
    # 额外优化
    torch_empty_cache_steps=50,  # 定期清空缓存
    
    # 减小 batch
    per_device_train_batch_size=1,
    gradient_accumulation_steps=16,  # 保持有效 batch size
    
    # 序列长度
    max_length=1024,  # 根据任务调整
    
    # Liger Kernel 加速 (需安装)
    use_liger_kernel=True,
)
```

### 4.2 训练稳定性优化

```python
from trl import GRPOConfig

config = GRPOConfig(
    # 梯度裁剪
    max_grad_norm=1.0,
    
    # 学习率调度
    lr_scheduler_type="cosine_with_min_lr",
    lr_scheduler_kwargs={"min_lr": 1e-7},
    
    # warmup
    warmup_ratio=0.1,
    
    # GRPO 特有
    scale_rewards="group",  # 更稳定的奖励缩放
)
```

### 4.3 日志与监控

```python
from trl import SFTConfig
from trl.trainer.callbacks import LogCompletionsCallback

config = SFTConfig(
    # 日志频率
    logging_steps=10,
    report_to=["wandb", "tensorboard"],
    
    # 保存策略
    save_strategy="steps",
    save_steps=500,
    save_total_limit=3,  # 只保留最近的 3 个 checkpoint
    
    # 评估
    eval_strategy="steps",
    eval_steps=100,
    load_best_model_at_end=True,
    metric_for_best_model="eval_loss",
)

# 添加生成日志回调 (GRPO)
trainer.add_callback(LogCompletionsCallback(
    log_completions_steps=50,
    num_completions=4,
))
```

## 5. 配置速查表

### SFT 关键参数

| 参数 | 推荐值 | 说明 |
|------|--------|------|
| `learning_rate` | 2e-5 (全量)<br>1e-4 (LoRA) | 学习率 |
| `max_length` | 2048 | 序列长度 |
| `packing` | True | 启用打包 |
| `packing_strategy` | "bfd" | 最佳适应递减 |
| `completion_only_loss` | True | 仅计算回复 loss |

### DPO 关键参数

| 参数 | 推荐值 | 说明 |
|------|--------|------|
| `learning_rate` | 1e-6 | 比 SFT 更低 |
| `beta` | 0.1 | KL 系数 |
| `loss_type` | ["sigmoid"] | 标准 DPO |
| `precompute_ref_log_probs` | False (GPU 充足)<br>True (GPU 受限) | 预计算 |

### GRPO 关键参数

| 参数 | 推荐值 | 说明 |
|------|--------|------|
| `num_generations` | 8 | 每组数量 |
| `max_completion_length` | 1024-2048 | 生成长度 |
| `num_iterations` | 2 | 每批更新 |
| `epsilon` | 0.2 | 裁剪下界 |
| `epsilon_high` | 0.28 | 裁剪上界 |
| `beta` | 0.001 | KL 系数 |
| `loss_type` | "dapo" | 推荐默认值 |
| `use_vllm` | True | 加速生成 |
| `vllm_gpu_memory_utilization` | 0.3 | vLLM 显存 |

## 6. 启动命令模板

### 单卡训练

```bash
# SFT
trl sft --model_name_or_path Qwen/Qwen2.5-0.5B \
    --dataset_name trl-lib/Capybara \
    --output_dir ./output/sft \
    --num_train_epochs 3 \
    --per_device_train_batch_size 4

# DPO
trl dpo --model_name_or_path ./output/sft \
    --dataset_name trl-lib/ultrafeedback_binarized \
    --output_dir ./output/dpo \
    --beta 0.1

# GRPO
trl grpo --model_name_or_path ./output/sft \
    --dataset_name trl-lib/DeepMath-103K \
    --reward_funcs accuracy_reward \
    --output_dir ./output/grpo \
    --num_generations 8 \
    --use_vllm
```

### 多卡训练

```bash
# DDP
torchrun --nproc_per_node=4 trl/scripts/sft.py \
    --model_name_or_path Qwen/Qwen2.5-0.5B \
    --dataset_name trl-lib/Capybara \
    --output_dir ./output/sft-multi

# DeepSpeed
accelerate launch --config_file accelerate_configs/deepspeed_zero3.yaml \
    trl/scripts/grpo.py \
    --model_name_or_path Qwen/Qwen2.5-0.5B \
    --dataset_name trl-lib/DeepMath-103K \
    --output_dir ./output/grpo-ds \
    --deepspeed ./ds_config_zero3.json
```
