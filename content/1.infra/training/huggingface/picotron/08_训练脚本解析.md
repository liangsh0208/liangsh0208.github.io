# 08_训练脚本解析

## 1. 目标论证

**为什么要解析训练脚本？**

train.py是picotron的入口，理解其流程有助于：
- 定位训练问题
- 修改训练流程(如添加评估、保存策略)
- 自定义数据加载和预处理

## 2. 工作假设

**配置假设：**
- 通过JSON配置文件传入所有参数
- 环境变量控制加速特性(FLASH_ATTEN等)

**可观测性：**
- 支持wandb日志记录
- 定期保存检查点
- 监控显存和吞吐量

## 3. 解决方案

### 3.1 主流程概览

```python
# train.py

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", required=True, help="Path to config.json")
    args = parser.parse_args()
    
    # 1. 加载配置
    config = load_config(args.config)
    
    # 2. 初始化分布式
    dist.init_process_group(backend="nccl")
    local_rank = int(os.environ["LOCAL_RANK"])
    device = torch.device(f"cuda:{local_rank}")
    torch.cuda.set_device(device)
    
    # 3. 配置日志
    setup_logging(config)
    
    # 4. 初始化4D进程组
    setup_process_group_manager(
        tp_size=config['distributed']['tp_size'],
        cp_size=config['distributed']['cp_size'],
        pp_size=config['distributed']['pp_size'],
        dp_size=config['distributed']['dp_size'],
    )
    
    # 5. 创建DataLoader
    data_loader = create_dataloader(config)
    
    # 6. 创建模型
    model = create_model(config, device)
    
    # 7. 创建优化器
    optimizer = create_optimizer(model, config)
    
    # 8. 恢复检查点(如有)
    start_step = load_checkpoint_if_exists(model, optimizer, config)
    
    # 9. 训练循环
    train_loop(model, data_loader, optimizer, config, start_step)
    
    # 10. 清理
    dist.destroy_process_group()
```

# create_config.py

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out_dir", required=True)
    parser.add_argument("--exp_name", required=True)
    parser.add_argument("--model_name", default="HuggingFaceTB/SmolLM-1.7B")

    parser.add_argument("--tp", type=int, default=1, help="Tensor Parallel size")
    parser.add_argument("--pp", type=int, default=1, help="Pipeline Parallel size")
    parser.add_argument("--cp", type=int, default=1, help="Context Parallel size")
    
    # Pipeline引擎
    parser.add_argument("--pp_engine", default="1f1b", choices=["1f1b", "afab"])
    
    # 训练配置
    parser.add_argument("--seq_len", type=int, default=1024)
    parser.add_argument("--mbs", type=int, default=4, help="Micro batch size")
    parser.add_argument("--grad_acc_steps", type=int, default=1)
    parser.add_argument("--max_steps", type=int, default=1000)
    parser.add_argument("--learning_rate", type=float, default=3e-4)
    
    # 模型覆盖
    parser.add_argument("--num_hidden_layers", type=int)
    
    args = parser.parse_args()
    
    config = {
        "distributed": {
            "tp_size": args.tp,
            "cp_size": args.cp,
            "pp_size": args.pp,
            "dp_size": args.dp,
            "pp_engine": args.pp_engine,
        },
        "model": {
            "name": args.model_name,
            "num_hidden_layers": args.num_hidden_layers,
        },
        "training": {
            "seq_length": args.seq_len,
            "micro_batch_size": args.mbs,
            "gradient_accumulation_steps": args.grad_acc_steps,
            "max_steps": args.max_steps,
            "learning_rate": args.learning_rate,
        },
        "environment": {
            "FLASH_ATTEN": os.getenv("FLASH_ATTEN", "0"),
            "HF_TOKEN": os.getenv("HF_TOKEN"),
        }
    }
    
    # 保存到实验目录

### 3.3 数据加载器

```python
class MicroBatchDataLoader:
    """支持micro-batch和梯度累积的DataLoader"""
    
    def __init__(self, config, dp_rank, dp_world_size):
        self.seq_length = config['seq_length']
        self.micro_batch_size = config['micro_batch_size']
        self.grad_acc_steps = config['gradient_accumulation_steps']
        
        # 全局batch = mbs * dp * grad_acc
        self.global_batch_size = (
            self.micro_batch_size * dp_world_size * self.grad_acc_steps
        )
        
        # DP分割(每个rank处理部分数据)
        self.dataset = load_dataset(config['dataset_name'])
        self.dataset = self.dataset.shard(
            num_shards=dp_world_size,
            index=dp_rank
        )
    
    def __iter__(self):
        for batch in self._batch_generator():
            yield {
                'input_ids': batch,
                'labels': batch
            }
    
    def _batch_generator(self):
        """生成固定长度的序列batch"""
        buffer = []
        
        for sample in self.dataset:
            tokens = tokenize(sample['text'])
            buffer.extend(tokens)
            
            while len(buffer) >= self.seq_length + 1:
                seq = buffer[:self.seq_length + 1]
                buffer = buffer[self.seq_length + 1:]
                
                if len(buffer_batch) < self.micro_batch_size:
                    buffer_batch.append(seq)
                else:
                    yield torch.tensor(buffer_batch)
                    buffer_batch = [seq]
```

### 3.4 训练步骤

```python
def train_step(model, data_loader, optimizer, device):
    """非PP模式的训练步骤"""
    batch = next(data_loader)
    input_ids = batch['input_ids'].to(device)
    labels = batch['labels'].to(device)
    
    # 前向
    outputs = model(input_ids=input_ids, labels=labels)
    loss = outputs['loss']
    
    # 反向
    loss.backward()
    
    return loss.item()


def train_step_pipeline_1f1b(model, data_loader, optimizer, device):
    """Pipeline模式训练步骤(1F1B调度)"""
    from picotron.pipeline_parallel import train_step_pipeline_1f1b as pipeline_train
    
    loss = pipeline_train(
        model,
        data_loader,
        device=device,
        grad_acc_steps=get_grad_acc_steps()
    )
    
    return loss


def train_loop(model, data_loader, optimizer, config, start_step):
    """主训练循环"""
    pgm = get_process_group_manager()
    device = next(model.parameters()).device
    
    # 是否使用pipeline
    use_pipeline = config['distributed']['pp_size'] > 1
    
    model.train()
    for step in range(start_step, config['training']['max_steps']):
        optimizer.zero_grad()
        
        step_losses = []
        
        # 梯度累积循环
        for accum_step in range(config['training']['gradient_accumulation_steps']):
            # 控制是否同步梯度(最后一步才同步)
            if hasattr(model, 'require_backward_grad_sync'):
                model.require_backward_grad_sync = (
                    accum_step == config['training']['gradient_accumulation_steps'] - 1
                )
            
            # 执行训练步骤
            if use_pipeline:
                loss = train_step_pipeline_1f1b(model, data_loader, optimizer, device)
            else:
                loss = train_step(model, data_loader, optimizer, device)
            
            loss = loss / config['training']['gradient_accumulation_steps']
            step_losses.append(loss)
        
        # 在DP+CP组内平均损失
        if pgm and pgm.cp_dp_world_size > 1:
            loss_tensor = torch.tensor([sum(step_losses)], device=device)
            dist.all_reduce(loss_tensor, group=pgm.cp_dp_group)
            avg_loss = loss_tensor.item() / pgm.cp_dp_world_size
        else:
            avg_loss = sum(step_losses)
        
        # 梯度裁剪
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        
        # 优化器步进
        optimizer.step()
        
        # 日志
        if pgm is None or pgm.global_rank == 0:
            print(f"Step {step}: loss = {avg_loss:.4f}")
            wandb.log({"train/loss": avg_loss, "step": step})
        
        # 保存检查点
        if step % config.get('save_every', 1000) == 0:
            save_checkpoint(model, optimizer, step)
```

## 4. 故障模式

| 问题 | 可能原因 | 解决方案 |
|-----|---------|---------|
| `RuntimeError: Address in use` | 端口冲突 | 设置 `MASTER_PORT` 环境变量 |
| Step stuck | 某个rank未到达同步点 | 检查所有rank的配置一致性 |
| 损失不下降 | 学习率过高或数据问题 | 降低lr，检查数据加载 |
| 显存突然增长 | 某个step激活未释放 | 使用torch.cuda.empty_cache() |
| Checkpoint保存失败 | 磁盘空间不足 | 检查 `/tmp` 或指定目录空间 |
