算力测试代码:

```
import torch
import time

def benchmark_gemm_flops(N=8192, dtype=torch.float16, device='cuda'):
    """
    测试单卡 FP16 GEMM 算力（TFLOPS）
    理论 FLOPs = 2 * N^3 （C = A @ B，A/B/C 为 NxN 矩阵）
    """
    if not torch.cuda.is_available():
        raise RuntimeError("CUDA not available")

    torch.cuda.set_device(0)
    torch.manual_seed(0)

    # 创建随机矩阵（确保不被优化掉）
    A = torch.randn(N, N, dtype=dtype, device=device)
    B = torch.randn(N, N, dtype=dtype, device=device)

    # 预热（warm-up）
    for _ in range(5):
        _ = torch.mm(A, B)

    torch.cuda.synchronize()
    start = time.time()

    # 正式计时（多次取平均）
    num_iter = 20
    for _ in range(num_iter):
        C = torch.mm(A, B)

    torch.cuda.synchronize()
    end = time.time()

    total_time = (end - start) / num_iter  # 平均每次时间（秒）
    flops = 2 * N**3 / total_time          # 单次 GEMM 的 FLOPs
    tflops = flops / 1e12

    print(f"Matrix size: {N}x{N}")
    print(f"Data type: {dtype}")
    print(f"Avg time: {total_time*1000:.2f} ms")
    print(f"Estimated TFLOPS: {tflops:.2f}")

if __name__ == "__main__":
    benchmark_gemm_flops(N=8192, dtype=torch.float16)
```
